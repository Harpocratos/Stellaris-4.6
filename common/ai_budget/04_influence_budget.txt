influence_expenditure_decisions = {
	resource = influence
	type = expenditure
	category = decisions

	potential = {
		always = yes
	}

	weight = {
		weight = 0.50
		modifier = {
			factor = 2
			OR = {
				AND = {
					has_technology = tech_housing_2
					any_owned_planet = {
						is_planet_class = pc_relic
						num_uncleared_blockers < 1
					}
				}
				AND = {
					has_ascension_perk = ap_arcology_project
					any_owned_planet = {
						is_artificial = no
						NOR = {
							is_planet_class = pc_city
							is_planet_class = pc_relic
							has_modifier = resort_colony
							has_modifier = penal_colony
							has_modifier = slave_colony
							has_modifier = pm_wenkwort_custodian
						}
						num_uncleared_blockers < 1
						free_district_slots < 1
						OR = {
							pop_amount >= 7000
							is_capital = yes
						}
					}
				}
			}
		}
	}

	# this is the cost of the arcology project decision
	desired_max = {
		base = 200
	}
}

influence_expenditure_starbases = {
	resource = influence
	type = expenditure
	category = starbases

	potential = {
		count_owned_nonprimary_starbase = {
			count > 0
		}
	}

	weight = {
		base = 0

		modifier = {
			add = 0.2
			has_technology = tech_housing_2
			any_owned_nonprimary_starbase = {
				exists = planet
				OR = {
					AND = {
						has_starbase_size = orbital_ring_tier_1
						count_starbase_modules = { count < 2 }
					}
					AND = {
						has_starbase_size = orbital_ring_tier_2
						count_starbase_modules = { count < 3 }
					}
					AND = {
						has_starbase_size = orbital_ring_tier_3
						count_starbase_modules = { count < 4 }
					}
				}
			}
		}
		modifier = {
			add = 0.3
			has_technology = tech_starbase_4
			any_owned_nonprimary_starbase = {
				has_starbase_size = orbital_ring_tier_1
				exists = planet
			}
		}
		modifier = {
			add = 0.3
			has_technology = tech_starbase_5
			any_owned_nonprimary_starbase = {
				has_starbase_size = orbital_ring_tier_2
				exists = planet
				planet = {
					OR = {
						has_planetary_ascension_tier >= 3
						has_major_upgraded_capital = yes
					}
				}
				OR = {
					count_starbase_modules = {
						count = 3
					}
					count_starbase_buildings = {
						count = 1
					}
				}
			}
		}
	}
	desired_min = {
		base = 50
	}
	desired_max = {
		base = 100
		modifier = {
			factor = 1.5
			has_technology = tech_starbase_5
			any_owned_nonprimary_starbase = {
				has_starbase_size = orbital_ring_tier_2
				exists = planet
				planet = {
					OR = {
						has_planetary_ascension_tier >= 3
						has_major_upgraded_capital = yes
					}
				}
			}
		}
	}
}

influence_expenditure_starbases_nomadic = {
	resource = influence
	type = expenditure
	category = starbases

	potential = {
		is_nomadic = yes
	}

	weight = {
		weight = 0.3
	}

	# Waystation tier upgrade = 50 influence. The arkship tier upgrade adds its 250 while one is available,
	# so the upgrade is funded before a new arkship is.
	desired_min = {
		base = @waystation_cost_influence_50
		modifier = {
			add = @arkship_cost_influence_250
			has_upgradable_arkship = yes
		}
	}

	# An Arkship tier upgrade and the Exodus Jump module each cost 250 influence in this category.
	# Reserve while either is still available to buy, and release once neither is.
	desired_max = {
		base = @waystation_cost_influence_50
		modifier = {
			add = @arkship_cost_influence_250
			OR = {
				AND = {
					has_technology = tech_arkship_tier_2
					any_owned_fleet = {
						starbase? = {
							is_arkship_starbase = yes
							can_be_upgraded = yes
						}
					}
				}
				AND = {
					has_technology = tech_arkship_exodus_jump
					any_owned_fleet = {
						starbase? = {
							arkship_spinal_slot_free = yes
						}
					}
				}
			}
		}
	}
}

influence_expenditure_operations = {
	resource = influence
	type = expenditure
	category = operations

	potential = {
		# only allocate influence for operations if we can afford them
		has_resource = {
			type = influence
			amount > 250
		}

		# and if we have a spy network to use them in!
		any_spynetwork = {
			owner = { is_same_value = root }
			is_spynetwork_level >= 20
		}
	}

	weight = {
		weight = 0.05

		modifier = {
			factor = 2
			has_tradition = tr_subterfuge_finish
		}

		modifier = {
			factor = 2
			any_spynetwork = {
				owner = { is_same_value = root }
				is_spynetwork_level >= 40
			}
		}

		modifier = {
			factor = 1.5
			any_spynetwork = {
				owner = { is_same_value = root }
				is_spynetwork_level >= 60
			}
		}

		modifier = {
			factor = 1.5
			any_spynetwork = {
				owner = { is_same_value = root }
				is_spynetwork_level >= 80
			}
		}
	}

	# based on gather information op
	desired_min = {
		base = 20
	}

	# based on crisis beacon op
	desired_max = {
		base = 320
	}
}
