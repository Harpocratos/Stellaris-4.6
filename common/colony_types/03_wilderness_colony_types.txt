# Heart World
col_wilderness_capital = {
	icon = GFX_colony_type_capital

	potential = {
		owner_or_space_owner? = { is_wilderness_empire = yes }
		is_capital = yes
	}

	planet_modifier = {
		planet_stability_add = 5
		planet_housing_add = 1000
		planet_crime_no_happiness_add = -20
		planet_jobs_produces_mult = 0.1
	}

	triggered_planet_modifier = {
		potential = {
			owner_or_space_owner? = { has_tradition = tr_adaptability_finish }
		}
		modifier = {
			planet_jobs_produces_mult = 0.05
		}
	}

	# Mirrors the other capitals: Archives (research) and Heavy Industry (foundry) alongside administrative buildings.
	ai_building_set_affinity = {
		government = 1.5
		research = 1.5
		physics = 1.25
		society = 1.25
		engineering = 1.25
		industrial = 1.5
		foundry = 1.5
	}

	weight_modifier = {
		base = 1000
	}
}

# Photosynthesis World
col_wilderness_generator = {
	icon = GFX_colony_type_generator

	potential = {
		is_special_colony_type = no
		is_capital = no
		owner_or_space_owner? = { is_wilderness_empire = yes }
	}

	planet_modifier = {
		planet_district_photosynthesis_fields_build_speed_mult = 0.25
		planet_technician_produces_mult = 0.25
	}

	triggered_planet_modifier = {
		potential = {
			owner_or_space_owner? = { has_tradition = tr_adaptability_finish }
		}
		modifier = { planet_technician_produces_mult = 0.1 }
	}
	triggered_planet_modifier = {
		potential = {
			owner_or_space_owner? = {
				is_galactic_community_member = yes
				is_gestalt = yes
			}
			OR = {
				is_active_resolution = "resolution_greatergood_five_year_plans"
				is_active_resolution = "resolution_greatergood_greater_than_ourselves"
				is_active_resolution = "resolution_greatergood_balance_in_the_middle"
				is_active_resolution = "resolution_greatergood_universal_prosperity_mandate"
			}
		}
		modifier = {
			planet_jobs_simple_drone_produces_mult = 0.1
		}
	}

	ai_building_set_affinity = {
		generator = 2.0
		generator_automation = 1.5
	}

	weight_modifier = {
		modifier = {
			add = @stickiness
			has_designation = col_wilderness_generator
		}
	}
}
