	size = large

	power = 0

	component_set = "paladin_crest"

	custom_tooltip = SHIP_PALADIN_CREST_TT

	scripted_action = { paladin_super_shield paladin_thunderclap paladin_recall paladin_air_superiority paladin_plan_b paladin_sudden_strike paladin_field_repairs paladin_taunt paladin_hold_the_line paladin_for_the_order paladin_teamwork paladin_leadership paladin_field_testing }

	friendly_aura = {
		name = "PALADIN_STRATEGIST_AURA"
		stack_info = {
			id = paladin_strategist_aura
			priority = 1
		}
		system_wide = yes
		limit = {
			solar_system? = {
				any_fleet_in_system = {
					controller = {
						OR = {
							is_friendly_to = root.controller

						}
					}
					exists = leader
					leader = {
						has_trait = leader_trait_paladin_strategist
					}
				}
			}
		}
		modifier = {
			ship_fire_rate_mult = 0.3
		}

				any_fleet_in_system = {
					controller = {
						OR = {
							is_friendly_to = root.controller
							is_same_value = root.controller
						}
					}
					exists = leader
					leader = {
						has_trait = leader_trait_paladin_logistician
					}
				}
			}
		}
		modifier = {

		}
	}

	ai_weight = {
		factor = 2
	}
