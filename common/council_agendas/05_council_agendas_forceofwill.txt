agenda_jubilee = {
	agenda_cost = @base_agenda_cost
	potential = {
		has_active_tenet = tenet_jubilation
		is_variable_set = jubilation_tier
	}

	init_effect = {
		custom_tooltip = agenda_jubilee_init_effect_tt
		hidden_effect = {
			jubilee_random_resource_roll = yes
			set_country_flag = has_selected_jubilee_agenda
		}
	}
	on_cancel = {
		jubilee_remove_modifiers = yes
		remove_country_flag = has_selected_jubilee_agenda
	}
	effect = {
		if = {
			limit = { NOT = { has_country_flag = has_selected_jubilee_agenda } }
			custom_tooltip = jubilee_finisher_tt
		}
		jubilee_finisher_effect = yes
		remove_country_flag = has_selected_jubilee_agenda
	}
}

agenda_show_of_force = {
	agenda_cost = @base_agenda_cost
	potential = {
		has_active_tenet = tenet_force_projection
	}

	init_effect = {
		if = {
			limit = {
				is_tenet_at_slot = {
					TENET = tenet_force_projection
					VALUE = 1
				}
			}
			custom_tooltip = agenda_show_of_force_unity_effect_tier_1
		}
		else_if = {
			limit = {
				is_tenet_at_slot = {
					TENET = tenet_force_projection
					VALUE = 2
				}
			}
			custom_tooltip = agenda_show_of_force_unity_effect_tier_2
		}
		else_if = {
			limit = {
				is_tenet_at_slot_or_higher = {
					TENET = tenet_force_projection
					VALUE = 3
				}
			}
			custom_tooltip = agenda_show_of_force_unity_effect_tier_3
		}
	}
	effect = {
		add_monthly_resource_mult = {
			resource = society_research
			value = @tier1researchreward
			min = @tier1researchmin
			max = @tier1researchmax
		}
	}
}
