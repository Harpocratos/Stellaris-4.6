# Build a Building
core_focus_100 = {
	category = core_focus
	tier = 1
	background = "GFX_focus_card_background_core_tier_1"

	name = "core.100.name"
	desc = "core.100.desc"
	hint = "core.100.hint"
	lore = "core.100.lore"

	condition_for_achievement = {
		custom_tooltip = {
			fail_text = "core.100.fail"
			success_text = "core.100.success"
			has_country_flag = planet_building_built
		}
	}

	on_success = {
		# Don't change the order of these three, as it corresponds to the card icon order
		add_focus_progress = { category = development amount = 10 }
		add_focus_progress = { category = conquest amount = 10 }
		add_focus_progress = { category = exploration amount = 10 }
		create_focus_completed_message = { CATEGORY = CORE }
	}
}

# Stockpile Energy Credits, Settled Empires
core_focus_118 = {
	base = core_focus_100

	name = "core.118.name"
	desc = "core.118.desc"
	hint = "core.118.hint"
	lore = "core.118.lore"

	possible = {
		is_nomadic = no
	}

	condition_for_achievement = {
		custom_tooltip = {
			fail_text = "core.118.fail"
			success_text = "core.118.success"
			if = {
				limit = {
					has_country_flag = focus_goal_stockpile_energy
				}
				custom_progress = {
					has_country_flag = focus_goal_stockpile_energy
					current_val_coeff = @focus_goal_stockpile_energy
					final_val_coeff = @focus_goal_stockpile_energy
				}
			}
			else = {
				custom_progress = {
					has_resource = {
						type = energy
						amount >= @focus_goal_stockpile_energy
					}
					mode = clamped
				}
			}
		}
	}
}
