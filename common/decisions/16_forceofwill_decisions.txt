decision_meditation = {
	icon = decision_meditation
	owned_planets_only = yes

	potential = {
		owner? = { is_path_of_the_self_empire = yes }
		NOT = { has_carrier_flag = decision_meditation_active }
	}

	effect = {
		set_carrier_flag = decision_meditation_active
		if = {
			limit = {
				owner = {
					has_country_flag = spirit_cultivation_body
				}
			}
			add_modifier = {
				modifier = planet_meditation_body
			}
		}
		else_if = {
			limit = {
				owner = {
					has_country_flag = spirit_cultivation_mind
				}
			}
			add_modifier = {
				modifier = planet_meditation_mind
			}
		}
		else = {
			add_modifier = {
				modifier = planet_meditation
			}
			custom_tooltip = decision_meditation_situation_effect
		}

		hidden_effect = {
			export_trigger_value_to_variable = {
				trigger = count_owned_pop_amount
				variable = local_pops
			}
			if = {
				limit = {
					check_variable = { which = local_pops value > 0 }
				}
				owner = {
					change_variable = {
						which = spirit_cultivation_planet_decision_pops
						value = local_pops
					}
				}
			}
		}
	}

	ai_weight = {
		weight = 0
		modifier = {
			add = 100
			owner = {
				OR = {
					has_country_flag = spirit_cultivation_free
					has_country_flag = spirit_cultivation_nerve
				}
			}
		}
		modifier = {
			add = 100
			owner = { has_country_flag = spirit_cultivation_body }
			meditation_body_housing_sufficient = yes
		}
		modifier = {
			add = 100
			owner = { has_country_flag = spirit_cultivation_mind }
			meditation_mind_amenities_sufficient = yes
		}
	}
}

						value = local_pops
					}
				}
			}
		}
	}
}

			add = 20
