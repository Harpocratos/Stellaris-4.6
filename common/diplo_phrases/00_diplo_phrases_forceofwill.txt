action_declare_ideological_rival = {
	select = {
		01_SELECT_DECLARE_IDEOLOGICAL_RIVAL = {}
		02_SELECT_DECLARE_IDEOLOGICAL_RIVAL = {}
	}
	propose = {
		01_PROPOSE_DECLARE_IDEOLOGICAL_RIVAL = {}
		01_PROPOSE_DECLARE_IDEOLOGICAL_RIVAL_SAME = {
			trigger = {
				ideology? = { has_same_ethic_as_ideology = from.ideology }
				from = {
					any_revealed_ideology = {
						is_same_value = root.ideology
					}
				}
			}
		}
		01_PROPOSE_DECLARE_IDEOLOGICAL_RIVAL_DIFFERENT = {
			trigger = {
				NOT = {
					ideology? = { has_same_ethic_as_ideology = from.ideology }
				}
				from = {
					any_revealed_ideology = {
						is_same_value = root.ideology
					}
				}
			}
		}
	}
	accept = {
		01_ACCEPT_DECLARE_IDEOLOGICAL_RIVAL = {}
		02_ACCEPT_DECLARE_IDEOLOGICAL_RIVAL = {}
	}
}
