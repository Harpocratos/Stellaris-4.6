planet_pops = {
	parent = planets
	generate_mult_modifiers = {
		produces
		upkeep
	}
	generate_add_modifiers = {
		upkeep
		produces
	}
	modifier_category = pop_group

	triggered_upkeep_modifier = {
		key = planet_pops_robotics
		modifier_types = {
			mult
		}
		trigger = {
			is_robot_pop_group = yes
		}
	}

	triggered_upkeep_modifier = {
		key = planet_pops_cyborgs
		modifier_types = {
			mult
		}
		trigger = {
			has_trait = trait_cybernetic
		}
	}

	triggered_upkeep_modifier = {
		key = planet_pops_organics
		modifier_types = {
			mult
		}
		trigger = {
			is_robot_pop_group = no
		}
	}

	triggered_upkeep_modifier = {
		key = planet_pops_unemployed
		modifier_types = {
			add
		}
		trigger = {
			is_unemployed = yes
		}
	}

	triggered_upkeep_modifier = {
		key = planet_pops_enslaved
		modifier_types = {
			mult
		}
		trigger = {
			is_enslaved = yes
		}
	}

	triggered_upkeep_modifier = {
		key = planet_pops_complex_drones
		modifier_types = {
			mult
		}
		trigger = {
			is_pop_category = complex_drone
		}
	}

	triggered_produces_modifier = {
		key = planet_pops_psionic
		modifier_types = {
			add
		}
		trigger = {
			has_psionic_species_trait = yes
		}
	}

	triggered_produces_modifier = {
		key = planet_pops_materialist_ideology
		modifier_types = {
			add
		}
		trigger = {
			any_followed_ideology = {
				is_materialist = yes
			}
		}
	}
}
