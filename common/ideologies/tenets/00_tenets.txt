		swap_type = {
			trigger = { is_machine_empire = yes }
			name = tenet_nepotism_machine
			desc = tenet_nepotism_machine_desc
		}
	}

	on_unlock = {
		if = {
			limit = { is_gestalt = no }
			if = {
				limit = { is_imperial_authority = no }
				custom_tooltip = tenet_nepotism_unlock_tt
			}
			else = {
				custom_tooltip = tenet_nepotism_imperial_unlock_tt
			}
			set_country_flag = tenet_nepotism_active
		}
	}
	on_lock = {
		if = {
			limit = { is_gestalt = no }
			remove_country_flag = tenet_nepotism_active
		}
	}

	tier = {
		# Regular

			potential = { is_gestalt = no }

			potential = { is_gestalt = yes }
			councilor_skill_add = 2
		}
	}
	tier = {
		# Regular
		triggered_country_modifier = {
			potential = { is_gestalt = no }
			councilor_skill_add = 2
		}
		# Gestalt
		triggered_country_modifier = {
			potential = { is_gestalt = yes }
			councilor_skill_add = 3
		}
