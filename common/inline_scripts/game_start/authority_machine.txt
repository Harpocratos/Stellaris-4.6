# Machine Intelligence Initial Setup
# Only for machine empires that are not nomadic
# Buildings Added:
#	* Administrative Array (no zone set)
#	* Machine Assembly Plants (no zone set)
#	* Drone Storage (no zone set)
#	* Civilian Industries (industry zone), Rogue Servitor or Obsessional
#	  Directive only
# Districts Added:
#	* 2 Central Nexus, planet size increased if needed
# Zones Added (Central Nexus):
#	* Slot 1: Archives
#	* Slot 2: Mixed Industry (Rogue Servitor or Obsessional Directive)
#	* Slot 2: Heavy Industry (all other machine empires)
# Other:
#	* Sets the rural district counts to 3 Generator, 4 Mining and 0 Agriculture
#	  Districts

if = {
	limit = {
		owner? = {
			is_machine_empire = yes
			is_nomadic = no
		}
	}
	add_building = building_machine_capital
	add_building = building_machine_assembly_plant
	add_building = building_drone_storage

	add_district_and_planet_size_if_needed_effect_amount = {
		amount = 2
		district = district_nexus
	}

	add_zone = {
		district = district_nexus
		zone = zone_research_unity
		zone_slot = 1
	}

	if = {
		limit = {
			owner? = {
				OR = {
					has_valid_civic = civic_machine_servitor
					has_valid_civic = civic_machine_obsessional_directive
				}
			}
		}
		add_zone = {
			district = district_nexus
			zone = zone_industrial
			zone_slot = 2
		}
		add_building = {
			building = building_factory_1
			zone = zone_industrial
		}
	}
	else = {
		add_zone = {
			district = district_nexus
			zone = zone_foundry
			zone_slot = 2
		}
	}

	set_variable = {
		which = energy_districts
		value = 3
	}
	set_variable = {
		which = mining_districts
		value = 4
	}
	set_variable = {
		which = farming_districts
		value = 0
	}
}
