# Anglers Initial Setup
# Only on Ocean worlds of Anglers empires
# Flags Set:
#	* uncapped_farming (carrier flag)
#	* angler_shackles (carrier flag), Broken Shackles only
# The following apply without the Broken Shackles origin
# Districts Added:
#	* 1 Agriculture District, planet size increased if needed
# Zones Added:
#	* Coastal Hamlets (Agriculture District), with Agrarian Idyll
#	* Aquaculture (Agriculture District), without Agrarian Idyll
# Other:
#	* Lowers the rural Agriculture District count by 1

if = {
	limit = {
		owner? = {
			is_anglers_empire = yes
			NOT = {
				has_origin = origin_broken_shackles
			}
		}
		is_planet_class = pc_ocean
	}
	set_carrier_flag = uncapped_farming

	add_district_and_planet_size_if_needed_effect_amount = {
		amount = 1
		district = district_farming
	}
	if = {
		limit = {
			owner? = {
				has_valid_civic = civic_agrarian_idyll
			}
		}
		add_zone = {
			district = district_farming
			zone = zone_agrarian_anglers
		}
	}
	else = {
		add_zone = {
			district = district_farming
			zone = zone_anglers
		}
	}
	# We`ve created a farming district, so reduce the count by 1
	change_variable = {
		which = farming_districts
		value = -1
	}
}
if = {
	limit = {
		owner? = {
			is_anglers_empire = yes
			has_origin = origin_broken_shackles
		}
		is_planet_class = pc_ocean
	}
	set_carrier_flag = uncapped_farming
	set_carrier_flag = angler_shackles
}
