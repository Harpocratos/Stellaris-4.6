# Environmentalist Initial Setup
# Not applied on artificial worlds or on
# Ecumenopolis, Hive, Machine and Relic Worlds
# Buildings Added:
#	* Ranger Lodge (research and unity zone)
# Cybernetic Creed sets no zone: the first zone with room that allows it
# Blockers Added:
#	* Nature Preserve
# Deposits Replaced:
#	* Industrial Wasteland, by a random natural blocker

if = {
	limit = {
		owner? = {
			has_valid_civic = civic_environmentalist
		}
		NOR = {
			is_planet_class = pc_city
			is_planet_class = pc_hive
			is_planet_class = pc_machine
			is_planet_class = pc_relic
			is_artificial = yes
		}
	}
	add_building_to_starting_research_unity_zone = {
		BUILDING = building_ranger_lodge
	}
	add_blocker = {
		type = d_big_nature_preserve_blocker
	}
	every_deposit = {
		limit = {
			is_deposit_type = d_failing_infrastructure
		}
		remove_deposit = yes
		prev = {
			add_random_natural_blocker = yes
		}
	}
}
