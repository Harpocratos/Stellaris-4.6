# Mutagenic Spas Initial Setup
# Buildings Added (industry zone):
#	* Mutagenic Spas for individualist empires and Rogue Servitors
#	* Mutagenic Permutation Pool for hive empires
#	* Hyper Lubrication Basin for other machine empires
# Cybernetic Creed sets no zone: the first zone with room that allows it

if = {
	limit = {
		owner? = {
			is_toxic_baths_empire = yes
			OR = {
				AND = {
					is_regular_empire = yes
					is_individual_machine = no
				}
				has_valid_civic = civic_machine_servitor
			}
		}
	}
	add_building_to_starting_industry_zone = {
		BUILDING = building_toxic_bath
	}
}

if = {
	limit = {
		owner? = {
			is_hive_empire = yes
			is_toxic_baths_empire = yes
		}
	}
	add_building_to_starting_industry_zone = {
		BUILDING = building_toxic_bath_hive
	}
}

if = {
	limit = {
		owner? = {
			NOT = {
				has_valid_civic = civic_machine_servitor
			}
			OR = {
				is_machine_empire = yes
				is_individual_machine = yes
			}
			is_toxic_baths_empire = yes
		}
	}
	add_building_to_starting_industry_zone = {
		BUILDING = building_toxic_bath_machine
	}
}
