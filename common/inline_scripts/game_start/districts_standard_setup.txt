# Standard Districts Setup
# Districts Added, by the parameter counts, planet size increased if needed:
#	* City District ($CITY$)
#	* Generator District ($GENERATOR$)
#	* Mining District ($MINING$)
#	* Agriculture District ($FARMING$)

while = {
	count = $CITY$
	add_district_and_planet_size_if_needed_effect = {
		district = district_city
	}
}
while = {
	count = $GENERATOR$
	add_district_and_planet_size_if_needed_effect = {
		district = district_generator
	}
}
while = {
	count = $MINING$
	add_district_and_planet_size_if_needed_effect = {
		district = district_mining
	}
}
while = {
	count = $FARMING$
	add_district_and_planet_size_if_needed_effect = {
		district = district_farming
	}
}
