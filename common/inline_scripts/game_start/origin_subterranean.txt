# Subterranean Initial Setup
# Also applied for Subterranean Machines
# Flags Set:
#	* uncapped_mining
# Districts Added:
#	* 1 Mining District, or 1 Magma Conduit on volcanic worlds
# Zones Added:
#	* Subterranean Urbanization on that district
# Rural Districts Removed:
#	* 1 Mining District, to offset the one added

if = {
	limit = {
		owner? = {
			is_subterranean_empire = yes
		}
	}
	if = {
		limit = {
			NOT = {
				is_planet_class = pc_volcanic
			}
		}
		set_carrier_flag = uncapped_mining
		add_district_and_planet_size_if_needed_effect_amount = {
			amount = 1
			district = district_mining
		}
		add_zone = {
			district = district_mining
			zone = zone_subterranean_urban
		}
		# We`ve created a mining district, so reduce the count by 1
		change_variable = {
			which = mining_districts
			value = -1
		}
	}
	else = { # Subterranean Volcanic World start
		set_carrier_flag = uncapped_mining
		add_district_and_planet_size_if_needed_effect_amount = {
			amount = 1
			district = district_melting
		}
		add_zone = {
			district = district_melting
			zone = zone_subterranean_urban
		}
		# We`ve created a mining district, so reduce the count by 1
		change_variable = {
			which = mining_districts
			value = -1
		}
	}
}
