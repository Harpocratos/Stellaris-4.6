# Espionage Support Cruise
	key = "CONTRACT_ESPIONAGE_SUPPORT_$ISSUER$_$VARIANT$_PROJECT"
	event_chain = contract_espionage_support_$ISSUER$_$VARIANT$_chain
	days_to_research = 30
	project_type = project_type_contract
	picture = $PICTURE$
	icon = "gfx/interface/icons/situation_log/situation_log_quest.dds"
	location = yes
	remove_when_completed = yes

	event_scope = ship_event

	requirements = {
		carries_colony = 1
	}

	on_success = {
		owner = {
			add_mission_counter = {
				mission = contract_espionage_support_$ISSUER$_$VARIANT$
				counter = contract_progress_counter
				amount = 1
			}
		}

		solar_system = {
			random_system_planet = {
				limit = {
					has_carrier_flag = spy_support_target@root
				}
				remove_carrier_flag = spy_support_target@root
			}
		}
	}

	on_fail = {
		owner = { end_event_chain = contract_espionage_support_$ISSUER$_$VARIANT$_chain }
	}

	abort_trigger = {
		NOT = {
			owner = { has_event_chain = contract_espionage_support_$ISSUER$_$VARIANT$_chain }
		}
	}

	on_cancel = {
		# Do nothing
	}
