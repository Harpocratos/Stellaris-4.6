#
# Mission categories
#
# Used to sort mission by types, and give general properties to each types
#
# The first category is the default one, if a mission is defined without category, it'll use that one
#
# Example :
# mission = {
#	is_contract = no	# whether this a contract or another type of mission
#	map_icon 			# path to the icon to use on the galactic map
#	log_icon			# path to the icon to use in the situation log (needs file path due to legacy code in the situation log)
#	show_in_issue_list = yes	# default = no, whether the category is shown at the top of the contract issue list (only applicable for contract categories)
# }
#
