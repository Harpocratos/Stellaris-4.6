# @opinion_failed_contract_regular = -125

## Espionage Support
contract_espionage_support = {
	category = diplomacy_contract
	event_chain = contract_espionage_support_chain
	icon = gfx/interface/icons/situation_log/situation_log_quest.dds
	picture = GFX_evt_decryption
	small_picture = GFX_contract_espionage_support_small
	location = yes
	time_to_accept = @contract_time_base
	cost = 500

	lore_issued = contract_issued_lore_espionage
	lore_completed = contract_completed_lore_espionage
	lore_failed = contract_failed_lore_espionage

	issued_abort_trigger = {
		is_nomadic = yes
	}

	abort_trigger = {
		NAND = {
			has_event_chain = contract_espionage_support_chain
			contract_target_system_reachable = {
				SYSTEM = fromfrom
			}
		}
	}

	counter = {
		contract_progress_counter = {
			max = 1
		}
	}

	potential_issuer = {
		can_issue_contracts_trigger = yes
		is_nomadic = no
		contract_system_is_owned_by_other_empire = yes
		fromfrom.owner = {
			num_owned_planets <= 3
		}
	}

	possible_issuer = {
		custom_tooltip = {
			fail_text = need_spy_network
			any_envoy = {
				has_envoy_task = {
					task = spy_network
					target = fromfrom.owner
				}
			}
		}
	}

	potential_operator = {
		NOT = { fromfrom = { is_at_war_with = root } }
	}

	possible_operator = {
		custom_tooltip = {
			fail_text = already_doing_same_contract
			NOT = {
				has_event_chain = contract_espionage_support_chain
			}
		}
		contract_target_system_reachable = {
			SYSTEM = from
		}
	}

	on_monthly = {
		country_event = { id = nomads.6000 }
	}

	on_issue = {
		set_country_flag = espionage_support_contract_active
	}

	on_accept = {
		begin_event_chain = {
			event_chain = contract_espionage_support_chain
			target = root
		}
		fromfrom = {
			random_system_planet = {
				limit = {
					is_colony = yes
				}
				set_carrier_flag = spy_support_target@root
				enable_special_project = { #at least one project
					name = "CONTRACT_ESPIONAGE_SUPPORT_PROJECT_01"
					location = this
					owner = root
				}
			}
		}
		prev = {
			random_spynetwork = {
				limit = {
					target = {
						is_same_value = fromfrom.owner
					}
				}
				set_spynetwork_flag = spy_support_contract_network@root
			}
		}
	}

	on_success = {
		if = {
			limit = {
				exists = root
			}
			espionage_support_effect = { INTEL_AMOUNT = 10 }
			custom_tooltip = trust10
			custom_tooltip = opinion100
			contract_scaling_reward_effect = { TRADE = 1750 RESOURCE = influence RESOURCE_AMOUNT = 35 }
			add_resource = { influence = @opinion_contract_regular }
			hidden_effect = {
				prev = { #should be issuer
					add_opinion_modifier = {
						modifier = opinion_completed_contract_regular
						who = root
					}
					add_trust = {
						amount = 10
						who = root
					}
				}
				country_event = { id = nomads.3002 days = 1 }
			}
		}
	}

	on_cancel = {
		if = {
			limit = {
				exists = root
			}
			prev = { #should be issuer
				add_opinion_modifier = {
					modifier = opinion_failed_contract_regular
					who = root
				}
			}
		}
	}

	on_stop = {
		if = {
			limit = {
				exists = root
			}
			end_event_chain = contract_espionage_support_chain
			espionage_support_cleanup_effect = yes
			prev = { remove_country_flag = espionage_support_contract_active }
		}
	}
}

	possible_issuer = {

			is_at_war = yes
		}
	}

	potential_operator = {
		NOT = {
			fromfrom = {
				OR = {
					is_at_war_with = root
					has_defensive_pact = root
					has_subject = root
					has_overlord = root
					is_in_federation_with = root
				}
			}
		}
	}

	possible_operator = {
		custom_tooltip = {
			fail_text = already_doing_same_contract
			NOT = {
				has_event_chain = contract_join_war_chain
			}
		}
		custom_tooltip = {
			fail_text = no_war_against_ally
			NOT = {
				any_country = {
					is_at_war_with = fromfrom
					OR = {
						has_non_aggression_pact = root
						has_defensive_pact = root
						has_subject = root
						has_overlord = root
						is_in_federation_with = root
					}
				}
			}
		}
		custom_tooltip = {
			fail_text = no_war_against_truce
			NOT = {
				any_country = {
					is_at_war_with = fromfrom
					has_truce = root
				}
			}
		}
		NOT = { #Cannot drag entire federations, or overlord to war over a contract
			has_federation = yes
			is_subject = yes
		}
	}

	on_monthly = {
		country_event = { id = nomads.6000 }
	}

	issued_abort_trigger = {
		OR = {
			is_nomadic = yes
			is_at_war = no
		}
	}

	abort_trigger = {
		OR = {
			NOT = {
				has_event_chain = contract_join_war_chain
			}
			prev = { is_at_war = no }
		}
	}

	counter = {
		contract_progress_counter = {
			max = 1
		}
	}

	on_issue = {
		# instant contract: cleaned up via nomads.3075/3076 when the war ends; the timer is a
		# safety net so the issuer is not left flagged if the contract is never picked up
		set_timed_country_flag = { flag = join_war_contract_active days = @contract_time_base }
	}

	on_accept = {
		begin_event_chain = {
			event_chain = contract_join_war_chain
			target = ROOT
		}
		#establish contact if need be
		if = {
			limit = {
				NOT = {
					has_communications = fromfrom.owner
				}
			}
			establish_communications = fromfrom.owner
		}
		#Join all of issuer's wars - using every_war + join_war_on_side to handle cases
		#where the issuer is a secondary war participant (e.g. joined via federation/defensive pact)
		#since join_war only works when the target is the main attacker/defender
		from = {
			save_event_target_as = join_war_contract_issuer
			every_war = {
				root = {
					join_war_on_side = {
						war = prev
						side = event_target:join_war_contract_issuer
					}
				}
			}
			set_country_flag = hired_nomad_merc@root
		}

		#finish contract
		add_mission_counter = {
			mission = contract_join_war
			counter = contract_progress_counter
			amount = 1
		}
	}

	on_success = {
		if = {
			limit = {
				exists = root
				OR = {
					has_valid_civic = civic_hired_guns
					has_valid_civic = civic_corporate_hired_guns
				}
			}
			contract_scaling_reward_effect = { TRADE = 25000 RESOURCE = alloys RESOURCE_AMOUNT = 5000 }
			# add hired_guns modifier
			add_modifier = {
				modifier = hired_guns_modifier
				days = -1
			}
		}
		else_if = {
			limit = { exists = root }
			contract_scaling_reward_effect = { TRADE = 16750 RESOURCE = alloys RESOURCE_AMOUNT = 3350 }
		}
		add_resource = { influence = @opinion_contract_large }
		custom_tooltip = trust15
		custom_tooltip = opinion150
		hidden_effect = {
			prev = { #should be issuer
				add_opinion_modifier = {
					modifier = opinion_completed_contract_large
					who = root
				}
				add_trust = {
					amount = 15
					who = root
				}
			}
			#flag wars for event tracking
			every_war = {
				limit = {
					any_war_participant = {
						is_same_value = from
					}
				}
				root = {
					set_country_flag = war@prev
				}
				set_war_flag = nomad_mercs
			}
			country_event = { id = nomads.3002 days = 1 }
		}
		#Event chain is ended when the war is ended
	}

	on_cancel = { #there shouldn't be time to cancel, but better safe than sorry
		if = {
			limit = {
				exists = root
			}
			prev = { #should be issuer
				add_opinion_modifier = {
					modifier = opinion_failed_contract_large
					who = root
				}
			}
		}
	}

	on_stop = {
		# This is an instant contract so no cleanup should happen here or it might break event logic. Instead cleanup is done via nomads.3075 and nomads.3080
	}
}

				}
				set_planet_flag = stellar_ignition_planet@root # root = operator; used by nomads.4500 to identify which planet completes the contract
				root = {
					create_point_of_interest = {
						id = stellar_ignition_poi@prev
						name = "stellar_ignition_poi_name"
						desc = stellar_ignition_poi_desc
						event_chain = contract_stellar_ignition_chain
						location = prev
					}
				}
			}
		}
	}

	on_success = {
		if = {
			limit = {
				exists = root
			}
			contract_scaling_reward_effect = { TRADE = 3250 RESOURCE = physics_research RESOURCE_AMOUNT = 650 }
			add_resource = { influence = @opinion_contract_regular }
		}
		else = {
			custom_tooltip = nomad_contract_stellar_ignition_tt
		}
		custom_tooltip = trust10
		custom_tooltip = opinion100
		hidden_effect = {
			prev = { #should be issuer
				add_opinion_modifier = {
					modifier = opinion_completed_contract_regular
					who = root
				}
				add_trust = {
					amount = 10
					who = root
				}
			}
			country_event = { id = nomads.3002 days = 1 }
		}
	}

	on_cancel = {
		if = {
			limit = {
				exists = root
			}
			prev = { #should be issuer
				remove_country_flag = stellar_ignition_contract_active
				add_opinion_modifier = {
					modifier = opinion_failed_contract_regular
					who = root
				}
			}
		}
	}

	on_stop = {
		if = {
			limit = {
				exists = root
			}
			end_event_chain = contract_stellar_ignition_chain
			prev = { remove_country_flag = stellar_ignition_contract_active }
			fromfrom = {
				random_system_planet = {
					limit = {
						has_planet_flag = stellar_ignition_planet@prevprevprevprev
					}
					remove_planet_flag = stellar_ignition_planet@prevprevprevprev
					remove_planet_flag = stellar_ignition_planet@root
					root = {
						remove_point_of_interest = stellar_ignition_poi@prev
					}
				}
			}
			remove_country_flag = successful_stellar_ignition@fromfrom
		}
	}
}

# The Eternal Chorus - self-issued discovery contract, spawned when Arkship enters a black hole system
contract_eternal_chorus = {
	category = science_contract_internal
	desc = "contract_eternal_chorus_desc"
	event_chain = arkship_eternal_chorus_chain
	icon = gfx/interface/icons/situation_log/situation_log_scanning_beam.dds
	picture = GFX_evt_black_hole
	location = yes
	time_to_accept = @contract_time_base
	cost = 0

	potential_issuer = {
		always = no
	}

	issued_abort_trigger = {
		OR = {
			is_nomadic = no
			NOT = {
				has_event_chain = arkship_eternal_chorus_chain
			}
		}
	}

	abort_trigger = {
		NOT = {
			has_event_chain = arkship_eternal_chorus_chain
		}
	}

	counter = {
		contract_progress_counter = { max = 1 }
	}

	on_accept = {
		random_system = {
			limit = { has_star_flag = eternal_chorus_system@ROOT }
			random_system_planet = {
				limit = { is_planet_class = pc_black_hole }
				save_event_target_as = eternal_chorus_black_hole
			}
		}
		enable_special_project = {
			name = "ARKSHIP_ETERNAL_CHORUS_PROJECT"
			location = event_target:eternal_chorus_black_hole
			owner = ROOT
		}
	}

	on_success = {
		add_monthly_resource_mult = {
			resource = trade
			value = @tier1materialreward
			min = @tier1materialmin
			max = @tier1materialmax
		}
		add_monthly_resource_mult = {
			resource = society_research
			value = @tier2researchreward
			min = @tier2researchmin
			max = @tier2researchmax
		}
		set_country_flag = arkship_eternal_chorus_completed
		random_system = {
			limit = { has_star_flag = eternal_chorus_system@ROOT }
			set_star_flag = eternal_chorus_origin@ROOT
		}
		hidden_effect = {
			country_event = { id = nomads.3002 days = 1 }
		}
	}

	on_cancel = {
		if = {
			limit = { exists = event_target:eternal_chorus_leader }
			event_target:eternal_chorus_leader = {
				remove_leader_flag = eternal_chorus_researcher
				remove_leader_flag = leader_death_events_blocked
			}
		}
	}

	on_stop = {
		remove_global_flag = eternal_chorus_active
		remove_country_flag = arkship_eternal_chorus_arrived
		clear_global_event_target = eternal_chorus_leader
		if = {
			limit = { has_special_project = ARKSHIP_ETERNAL_CHORUS_PROJECT }
			abort_special_project = { type = ARKSHIP_ETERNAL_CHORUS_PROJECT }
		}
		random_system = {
			limit = { has_star_flag = eternal_chorus_system@ROOT }
			remove_star_flag = eternal_chorus_system@ROOT
			random_system_planet = {
				limit = { has_planet_flag = eternal_chorus_planet@ROOT }
				remove_planet_flag = eternal_chorus_planet@ROOT
			}
		}
		end_event_chain = arkship_eternal_chorus_chain
	}
}

	cost = 0

	potential_issuer = {
		always = no
	}

	abort_trigger = {
		NOT = {
			has_event_chain = arkship_diplomatic_envoy_chain
		}
	}

	counter = {
		contract_progress_counter = { max = 1 }
	}

	on_accept = {
		begin_event_chain = {
			event_chain = arkship_diplomatic_envoy_chain
			target = ROOT
		}
		random_system = {
			limit = { has_star_flag = diplomatic_envoy_target_system@ROOT }
			ROOT = {
				create_point_of_interest = {
					id = diplomatic_envoy_target_poi
					name = "diplomatic_envoy_target_poi_name"
					desc = "diplomatic_envoy_target_poi_desc"
					event_chain = arkship_diplomatic_envoy_chain
					location = prev
				}
			}
		}
		every_owned_ship = {
			limit = {
				is_arkship_ship = yes
				solar_system = {
					has_star_flag = diplomatic_envoy_target_system@ROOT
				}
			}
			carrier_event = { id = arkship.160 }
		}
	}

	on_success = {
		add_monthly_resource_mult = {
			resource = influence
			value = @tier2influencereward
			min = @tier2influencemin
			max = @tier2influencemax
		}
		add_monthly_resource_mult = {
			resource = trade
			value = @tier1materialreward
			min = @tier1materialmin
			max = @tier1materialmax
		}
		set_country_flag = arkship_diplomatic_envoy_completed
		custom_tooltip = trust5
		custom_tooltip = opinion50
		hidden_effect = {
			random_country = {
				limit = { has_country_flag = diplomatic_envoy_target_of@ROOT }
				add_opinion_modifier = {
					modifier = opinion_completed_contract_small
					who = root
				}
				add_trust = {
					amount = 5
					who = root
				}
			}
			country_event = { id = nomads.3002 days = 1 }
		}
	}

	on_cancel = {
	}

	on_stop = {
		remove_country_flag = arkship_diplomatic_envoy_arrived
		random_owned_leader = {
			limit = { has_leader_flag = diplomatic_envoy_official }
			remove_leader_flag = diplomatic_envoy_official
			remove_leader_flag = leader_death_events_blocked
		}
		random_country = {
			limit = { has_country_flag = diplomatic_envoy_target_of@ROOT }
			remove_country_flag = diplomatic_envoy_target_of@ROOT
		}
		random_system = {
			limit = { has_star_flag = diplomatic_envoy_target_system@ROOT }
			remove_star_flag = diplomatic_envoy_target_system@ROOT
		}
		remove_point_of_interest = diplomatic_envoy_target_poi
		end_event_chain = arkship_diplomatic_envoy_chain
	}
