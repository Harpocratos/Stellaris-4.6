##############################
# Imperialist Faction
##############################

imperialist = {
	guiding_ethic = ethic_militarist
	swappable_data = {
		default = {
			visual_ethic = imperialist_visual_ethic
		}
		swap_type = {
			trigger = { is_faction_fanatic = yes }
			visual_ethic = ethic_fanatic_militarist_texticon
		}
	}

	inline_script = {
		script = pop_faction_types/ideological_swap
		ETHIC = militarist
	}

	unique = yes
	support_multiplier = 1

	resources = {
		category = pop_factions
		inline_script = {
			script = pop_faction_types/resources/unity_trade_research
			MAJOR_TECH = engineering
			MINOR_TECH = society
		}
	}

	extortion_resource = {
		category = pop_factions
		produces = {
			trigger = {
				owner = {
					is_crowdsourcing_empire = no
				}
			}
			alloys = @faction_base_output
			mult = @faction_extortion_alloys_ratio
		}
		produces = {
			trigger = {
				owner = {
					is_crowdsourcing_empire = yes
				}
			}
			engineering_research = @faction_major_output
			society_research = @faction_minor_output
			mult = @faction_extortion_crowdsourcing_ratio
		}
	}

	#Note: number of potential or actual members is checked by POP_FACTION_MIN_POTENTIAL_MEMBERS define
	is_potential = {
		is_faction_allowed = yes
		NOT = { has_country_flag = suppressed_imperialist_destroyed }
	}

	valid = {
		is_faction_valid = yes
		is_ideology_leftover_faction = no
	}

	can_join_pre_triggers = {
		is_enslaved = no
		is_being_purged = no
	}

	can_join_faction = {
		inline_script = pop_faction_types/can_join_faction/base
	}

	pop_attraction_tag = {
		desc = POP_ATTRACTION_AUTHORITARIAN_POS
		trigger = {
			NOT = { exists = ideology }
			owner? = {
				is_authoritarian = yes
			}
		}
	}
	pop_attraction_tag = {
		desc = POP_FACTION_ATTRACTION_CITIZENSHIP_POS
		trigger = {
			NOT = { exists = ideology }
			owner? = {
				any_owned_pop_group = {
					has_citizenship_rights = yes
				}
			}
		}
	}

	attraction = {
		base = 100

		inline_script = pop_faction_types/attraction/ideological_attraction
		inline_script = pop_faction_types/attraction/leader_attraction

		modifier = {
			factor = 1.25
			desc = OWNER_IS_AUTHORITARIAN
			NOT = { from? = { exists = ideology } }
			owner? = { is_authoritarian = yes }
		}

		modifier = {
			factor = 1.25
			NOT = { from? = { exists = ideology } }
			has_citizenship_rights = yes
		}
	}

	leader = {
		base = 100

		modifier = {
			factor = 0.0
			is_event_leader = yes
		}
	}

	on_create = {
		pop_faction_event = { id = factions.0 }
		owner = { remove_country_flag = suppressed_imperialist_destroyed }
	}

	on_destroy = {
		inline_script = {
			script = pop_faction_types/effect/promote_on_destroy
			ETHIC = militarist
		}
		inline_script = {
			script = pop_faction_types/effect/suppress_on_destroy_base
			ETHIC = militarist
			TYPE = imperialist
		}
		inline_script = {
			script = pop_faction_types/effect/suppress_on_destroy_all
			ETHIC = militarist
		}
	}

	# Demands
	inline_script = {
		script = pop_faction_types/demands/imperialist
	}

	# Actions
	inline_script = {
		script = pop_faction_types/actions/advocate_faction
	}
	inline_script = {
		script = pop_faction_types/actions/embrace_faction
		ETHIC = militarist
	}
	inline_script = {
		script = pop_faction_types/actions/promote_faction
		ETHIC = militarist
	}
	inline_script = {
		script = pop_faction_types/actions/suppress_faction
		ETHIC = militarist
	}
	inline_script = {
		script = pop_faction_types/actions/extort_faction_crowdsourcing
		RESOURCE = alloys
		CROWDSOURCE_TT = crowdsourced_extortion_tt
	}
}
