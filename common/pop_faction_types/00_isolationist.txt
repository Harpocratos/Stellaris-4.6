##############################
# Isolationist Faction
##############################

isolationist = {
	guiding_ethic = ethic_xenophobe
	swappable_data = {
		default = {
			visual_ethic = isolationist_visual_ethic
		}
		swap_type = {
			trigger = { is_faction_fanatic = yes }
			visual_ethic = ethic_fanatic_xenophobe_texticon
		}
	}

	inline_script = {
		script = pop_faction_types/ideological_swap
		ETHIC = xenophobe
	}

	unique = yes
	support_multiplier = 1

	resources = {
		category = pop_factions
		inline_script = {
			script = pop_faction_types/resources/unity_trade_research
			MAJOR_TECH = society
			MINOR_TECH = engineering
		}
	}

	extortion_resource = {
		category = pop_factions
		produces = {
			trigger = {
				owner = {
					is_crowdsourcing_empire = no
				}
			}
			minerals = @faction_base_output
			mult = @faction_extortion_minerals_ratio
		}
		produces = {
			trigger = {
				owner = {
					is_crowdsourcing_empire = yes
				}
			}
			society_research = @faction_major_output
			engineering_research = @faction_minor_output
			mult = @faction_extortion_crowdsourcing_ratio
		}
	}

	is_potential = {
		is_faction_allowed = yes
		NOT = { has_country_flag = suppressed_isolationist_destroyed }
	}

	valid = {
		is_faction_valid = yes
		is_ideology_leftover_faction = no
	}

	can_join_pre_triggers = {
		is_enslaved = no
		is_being_purged = no
	}

	can_join_faction = {
		inline_script = pop_faction_types/can_join_faction/base
		OR = {
			NOT = { is_same_species = owner_species }
			custom_tooltip_icon = {
				text = OWNER_IS_PACIFIST
				owner? = {
					is_pacifist = yes
				}
			}
		}
	}

	pop_attraction_tag = {
		desc = POP_ATTRACTION_PACIFIST_POS
		trigger = {
			NOT = { exists = ideology }
			owner? = {
				is_pacifist = yes
			}
		}
	}

	attraction = {
		base = 100

		inline_script = pop_faction_types/attraction/ideological_attraction
		inline_script = pop_faction_types/attraction/leader_attraction

		modifier = {
			factor = 1.25
			desc = OWNER_IS_PACIFIST
			NOT = { from? = { exists = ideology } }
			owner? = { is_pacifist = yes }
		}
	}

	leader = {
		base = 100

		modifier = {
			factor = 0.0
			is_event_leader = yes
		}
	}

	on_create = {
		pop_faction_event = { id = factions.0 }
		owner = { remove_country_flag = suppressed_isolationist_destroyed }
	}

	on_destroy = {
		inline_script = {
			script = pop_faction_types/effect/promote_on_destroy
			ETHIC = xenophobe
		}
		inline_script = {
			script = pop_faction_types/effect/suppress_on_destroy_base
			ETHIC = xenophobe
			TYPE = isolationist
		}
		inline_script = {
			script = pop_faction_types/effect/suppress_on_destroy_all
			ETHIC = xenophobe
		}
	}

	# Demands
	inline_script = {
		script = pop_faction_types/demands/isolationist
	}

	# Actions
	inline_script = {
		script = pop_faction_types/actions/advocate_faction
	}
	inline_script = {
		script = pop_faction_types/actions/embrace_faction
		ETHIC = xenophobe
	}
	inline_script = {
		script = pop_faction_types/actions/promote_faction
		ETHIC = xenophobe
	}
	inline_script = {
		script = pop_faction_types/actions/suppress_faction
		ETHIC = xenophobe
	}
	inline_script = {
		script = pop_faction_types/actions/extort_faction_crowdsourcing
		RESOURCE = minerals
		CROWDSOURCE_TT = crowdsourced_extortion_tt
	}
}
