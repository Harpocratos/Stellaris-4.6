##############################
# Xenoist Faction
##############################

xenoist = {
	guiding_ethic = ethic_xenophile
	swappable_data = {
		default = {
			visual_ethic = xenoist_visual_ethic
		}
		swap_type = {
			trigger = { is_faction_fanatic = yes }
			visual_ethic = ethic_fanatic_xenophile_texticon
		}
	}

	inline_script = {
		script = pop_faction_types/ideological_swap
		ETHIC = xenophile
	}

	unique = yes
	support_multiplier = 1

	resources = {
		category = pop_factions
		inline_script = {
			script = pop_faction_types/resources/unity_trade_research
			MAJOR_TECH = society
			MINOR_TECH = physics
		}
	}

	extortion_resource = {
		category = pop_factions
		produces = {
			trigger = {
				owner = {
					is_crowdsourcing_empire = no
				}
			}
			energy = @faction_base_output
			mult = @faction_extortion_energy_ratio
		}
		produces = {
			trigger = {
				owner = {
					is_crowdsourcing_empire = yes
				}
			}
			society_research = @faction_major_output
			physics_research = @faction_minor_output
			mult = @faction_extortion_crowdsourcing_ratio
		}
	}

	is_potential = {
		is_faction_allowed = yes
		NOT = { has_country_flag = suppressed_xenoist_destroyed }
	}

	valid = {
		is_faction_valid = yes
		is_ideology_leftover_faction = no
	}

	can_join_pre_triggers = {
		is_enslaved = no
		is_being_purged = no
	}

	can_join_faction = {
		inline_script = pop_faction_types/can_join_faction/base
	}

	pop_attraction_tag = {
		desc = POP_ATTRACTION_EGALITARIAN_POS
		trigger = {
			NOT = { exists = ideology }
			owner? = {
				is_egalitarian = yes
			}
		}
	}
	pop_attraction_tag = {
		desc = POP_FACTION_ATTRACTION_XENO_POS
		trigger = {
			NOT = { exists = ideology }
			owner? = {
				any_owned_species = {
					NOT = {
						is_same_species = prev
					}
				}
			}
		}
	}

	attraction = {
		base = 150

		inline_script = pop_faction_types/attraction/ideological_attraction
		inline_script = pop_faction_types/attraction/leader_attraction

		modifier = {
			factor = 1.25
			desc = OWNER_IS_EGALITARIAN
			NOT = { from? = { exists = ideology } }
			owner? = { is_egalitarian = yes }
		}

		modifier = {
			factor = 2.0
			NOT = { from? = { exists = ideology } }
			NOT = { is_same_species = owner_species }
		}
	}

	leader = {
		base = 100

		modifier = {
			factor = 0.5
			leader_class = scientist
		}

		modifier = {
			factor = 2.0
			NOT = { is_same_species = owner_species }
		}

		modifier = {
			factor = 0.0
			is_event_leader = yes
		}
	}

	on_create = {
		pop_faction_event = { id = factions.0 }
		owner = { remove_country_flag = suppressed_xenoist_destroyed }
	}

	on_destroy = {
		inline_script = {
			script = pop_faction_types/effect/promote_on_destroy
			ETHIC = xenophile
		}
		inline_script = {
			script = pop_faction_types/effect/suppress_on_destroy_base
			ETHIC = xenophile
			TYPE = xenoist
		}
		inline_script = {
			script = pop_faction_types/effect/suppress_on_destroy_all
			ETHIC = xenophile
		}
	}

	# Demands
	inline_script = {
    	script = pop_faction_types/demands/xenoist
    }

	# Actions
	inline_script = {
		script = pop_faction_types/actions/advocate_faction
	}
	inline_script = {
		script = pop_faction_types/actions/embrace_faction
		ETHIC = xenophile
	}
	inline_script = {
		script = pop_faction_types/actions/promote_faction
		ETHIC = xenophile
	}
	inline_script = {
		script = pop_faction_types/actions/suppress_faction
		ETHIC = xenophile
	}
	inline_script = {
		script = pop_faction_types/actions/extort_faction_crowdsourcing
		RESOURCE = energy
		CROWDSOURCE_TT = crowdsourced_extortion_tt
	}
}
