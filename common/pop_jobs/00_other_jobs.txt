civilian = {
	category = civilian
	is_capped_by_modifier = no
	count_as_available_for_ai = yes

	possible_pre_triggers = {
		has_owner = yes
		is_enslaved = no
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}

	possible = {
		hidden_trigger = { exists = owner }
		owner = {
			is_gestalt = no
		}
		NOT = {
			has_citizenship_type = {
				type = citizenship_cruise_passenger
				country = owner
			}
		}
		has_disconnected_drone_citizenship_type = no
		OR = {
			NOT = {
				owner = {
					has_active_tradition = tr_virtuality_finish
				}
			}
			has_virtual_species_trait = no
		}
		colony? = { NOT = { is_planet_class = pc_cosmogenesis_world }  }
	}

	swappable_data = {
		default = {
			building_icon = building_residence
		}
		# Tourists
		swap_type = {
			trigger = {
				exists = owner
				has_modifier = resort_colony
			}
			name = tourist
			desc = tourist_desc
			icon = trader
			building_icon = building_amusement_megaplex
			weight = 6
		}
		# Hedonists
		swap_type = {
			trigger = {
				owner? = {
					OR = {
						is_pleasure_seeker_empire = yes
						has_active_tenet = tenet_hedonism
					}
				}
			}
			name = fe_hedonist
			desc = fe_hedonist
			icon = fe_hedonist
			building_icon = building_holo_theatres
			weight = 5
		}
		# Faithful Acolyte
		swap_type = {
			trigger = {
				civilian_is_sacred_path_civilian = yes
			}
			name = sacred_path_civilian
			desc = job_sacred_path_civilian_desc
			icon = sacred_path_civilian
			building_icon = building_temple
			weight = 10
		}
	}

	resources = {
		category = planet_civilians
		produces = {
			trade = 1
		}
		produces = {
			trigger = {
				exists = owner
				has_modifier = resort_colony
			}
			trade = 1
		}
		# Hedonists
		produces = {
			trigger = {
				owner? = { is_pleasure_seeker_empire = yes }
			}
			unity = 1
		}
		upkeep = {
			trigger = {
				owner? = { is_pleasure_seeker_empire = yes }
			}
			consumer_goods = 0.25
		}
		inline_script = "jobs/ethic_job_resources"
	}

	# Hedonists
	triggered_planet_modifier = {
		potential = {
			owner? = {
				is_pleasure_seeker_empire = no
				NOR = {
					has_valid_civic = civic_dystopian_society
					has_active_tenet = tenet_hedonism
				}
			}
		}
		planet_amenities_add = 50
		mult = value:scripted_modifier_mult|MODIFIER|pop_job_amenities_mult|
	}

	triggered_country_modifier = {
		potential = {
			owner? = { has_modifier = paragon_origin_strengthened_government_iii }
		}
		country_edict_fund_add = 1
	}

	inline_script = "jobs/ethic_job_modifiers"

	# Hedonism tenet
	triggered_planet_modifier = {
		potential = {
			owner? = {
				has_active_tenet = tenet_hedonism
				is_synthetic_empire = no
			}
		}
		logistic_growth_mult = 1
		mult = owner.modifier:tenet_hedonism_logistic_growth_modifier
	}
	triggered_planet_modifier = {
		potential = {
			owner? = {
				has_active_tenet = tenet_hedonism
				is_synthetic_empire = yes
			}
		}
		roboticist_jobs_bonus_workforce_mult = 1
		mult = owner.modifier:tenet_hedonism_roboticist_workforce_modifier
	}

	triggered_planet_modifier = {
		potential = {
			owner? = {
				has_active_tenet = tenet_militia_conscription
			}
		}
		planet_defense_armies_add = 1
		mult = owner.modifier:tenet_militia_conscription_defense_army_modifier
	}

	triggered_country_modifier = {
		potential = {
			owner? = {
				has_active_tenet = tenet_militia_conscription
				is_nomadic = yes
			}
		}
		shipclass_starbase_hull_mult = 1
		mult = owner.modifier:tenet_militia_conscription_arkship_hull_modifier
	}

	weight = {
		weight = 0
	}
}
