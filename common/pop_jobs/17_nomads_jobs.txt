cruise_passenger = {
	category = cruise_passenger
	can_be_automated = no

	swappable_data = {
		default = {
			building_icon = building_organic_sanctuary
		}
		swap_type = {
			trigger = {
				owner? = { is_hospitality_servitor_empire = yes }
			}
			name = cruise_passenger_bio_trophy
			building_icon = building_organic_sanctuary
		}
	}
	tags = { unity cruise_passenger }

	possible_pre_triggers = {
		is_sapient = yes
	}

	possible = {
		owner = { has_origin = origin_forever_cruise }
		OR = {
			has_living_standard = { type = living_standard_cruise_passenger country = owner }
			has_living_standard = { type = living_standard_organic_trophy country = owner }
		}
	}

	resources = {
		category = planet_cruise_passengers
		produces = {
			unity = 2
		}
		upkeep = {
			trigger = {
				owner? = { is_hospitality_servitor_empire = no }
			}
			consumer_goods = 1.5
		}
		upkeep = {
			trigger = {
				owner? = { is_hospitality_servitor_empire = yes }
			}
			consumer_goods = 1
		}
	}

	triggered_planet_modifier = {
		potential = {
			owner? = { is_gestalt = no }
		}
		pop_cat_specialist_bonus_workforce_mult = 0.01
	}

	triggered_planet_modifier = {
		potential = {
			owner? = { is_gestalt = yes }
		}
		pop_cat_complex_drone_bonus_workforce_mult = 0.01
	}

	# Hospitality Servitors pamper their Bio-Trophy Passengers, granting an additional Complex Drone Job Efficiency bonus on top of the base Gestalt amount.
	triggered_planet_modifier = {
		potential = {
			owner? = { is_hospitality_servitor_empire = yes }
		}
		pop_cat_complex_drone_bonus_workforce_mult = 0.01
	}

	weight = {
		weight = 9999
	}

	inline_script = "jobs/automodding_priority_charisma"
}

cruise_crew_overseer = { #Based off resort worker
	category = ruler

	tags = { unity amenities }

	swappable_data = {
		default = {
			condition_string = RULER_JOB_TRIGGER
			building_icon = building_forever_cruise_crew_service
		}
	}

	possible_pre_triggers = {
		has_owner = yes
		is_being_purged = no
		is_being_assimilated = no
	}

	possible_precalc = can_fill_ruler_job

	possible = {
		entertainer_job_check_trigger = yes
		owner = { has_origin = origin_forever_cruise }
	}

	resources = {
		category = planet_entertainers
		produces = {
			unity = 2
			trade = 2
		}
		upkeep = {
			consumer_goods = 3
		}
	}

	triggered_planet_modifier = {
		potential = {
			owner? = { has_tradition = tr_mercantile_commercial_enterprise }
		}
		planet_amenities_add = 100
		mult = value:scripted_modifier_mult|MODIFIER|pop_job_amenities_mult|
	}

	planet_modifier = {
		pop_cat_worker_bonus_workforce_mult = 0.05
		pop_cat_specialist_bonus_workforce_mult = 0.025
	}

	weight = {
		weight = @ruler_job_weight
		# crisis purge
		modifier = {
			factor = 0.01
			colony.controller? = {
				OR = {
					is_country_type = swarm
					is_country_type = ai_empire
				}
			}
			OR = { # contingency won't purge itself
				NOT = { exists = event_target:custodian_bot }
				AND = {
					exists = event_target:custodian_bot
					NOT = { species = { is_same_value = event_target:custodian_bot } }
				}
			}
		}
	}
}
