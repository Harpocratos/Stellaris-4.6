# For Modifiers :
# Root = Origin Country
# From = Origin Country's founder species
# This = Generated Ideology

# Militarist
ideology_name_format = {
	random_weight = {
		factor = 0
		modifier = {
			add = 1
			is_militarist = yes
			root = {
				NOT = { has_origin = origin_lost_makers }
			}
		}
	}
	format = "{ideology_ajd_keyword {[From.GetAdj] <militarist_ideology_names>}}"
