## Empower the council
resolution_custodian_council = {
	icon = "GFX_resolution_custodian_council"

	resources = {
		category = resolutions
		cost = {
			influence = @resolution_cost_t5
		}
	}
	target = no

	triggered_modifier = {
		potential = {
			is_part_of_galactic_council = yes
		}
		modifier = {
			custom_tooltip = resolution_custodian_council_tooltip
			resolution_custodian_council = 1
		}
	}

	fail_effects = {
	}

	potential = {
		is_galactic_custodian = yes
		has_menace_perk = menp_defender_custodian
		has_global_flag = galactic_council_formed
		num_council_positions > 1
	}

	allow = {
		custom_tooltip = {
			fail_text = "resolution_already_active"
			NOT = { is_active_resolution = "resolution_custodian_council" }
		}
	}

	ai_weight = {
		base = @resolution_weight_normal

		modifier = {
			factor = 1.5
			OR = {
				has_civic = civic_diplomatic_corps
				has_civic = civic_public_relations_specialists
				has_civic = civic_hive_empath
				has_civic = civic_machine_servitor
				has_civic = civic_machine_exploration_protocol
			}
			desc = gal_com_opposing_civics
		}

		modifier = {
			factor = 0.8
			is_galactic_community_member = yes
			desc = gal_com_negative_sentiment
		}

		modifier = {
			factor = 1.2
			is_egalitarian = yes
			desc = ethic_egalitarian
		}

		modifier = {
			factor = 2
			OR = {
				is_part_of_galactic_council = yes
				AND = {
					is_subject = yes
					overlord = {
						opinion_level = { who = prev level >= neutral }
						is_part_of_galactic_council = yes
					}
				}
			}
			desc = gal_com_they_or_overlord_on_galatic_council
		}

		modifier = {
			factor = 0.8
			galactic_community_rank > 5
			NAND = {
				is_subject = yes
				overlord = {
					opinion_level = { who = prev level >= neutral }
					is_part_of_galactic_council = yes
				}
			}
			desc = gal_com_they_nor_overlord_not_on_galatic_council
		}
	}
}
