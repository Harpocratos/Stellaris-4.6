# Per-fleet target template size for create_random_fleet. With 1:4 large:small weights, slider 1.0x yields ~2-3 large + ~10 small per fleet (~210-260 nav cap, varies run-to-run).
# Slider: fallen_empire_strength_scale (galaxy setup), linear multiplier 0.25x to 5.0x.
fe_starting_fleet_size = {
	base = 200
	complex_trigger_modifier = {
		trigger = check_galaxy_setup_value
		parameters = {
			setting = fallen_empire_strength_scale
		}
		mode = mult
	}
}
