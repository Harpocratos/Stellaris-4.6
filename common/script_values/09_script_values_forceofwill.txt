# One Cultural Envoy per official the empire can spare to crew one, and none at all until it can pay for the ship
# twice over. Officials already crewing an envoy are counted, so the fleet does not shrink back once they are manned.
desired_cultural_envoys = {
	base = 0
	complex_trigger_modifier = {
		trigger = count_owned_leader
		parameters = {
			limit = {
				leader_class = official
				OR = {
					AND = {
						is_idle = yes
						is_councilor = no
					}
					fleet? = {
						is_ship_class = shipclass_cultural_envoy
					}
				}
			}
		}
		mode = add
	}
	modifier = {
		factor = 0
		OR = {
			resource_stockpile_compare = {
				resource = influence
				value < @[ cultural_envoy_cost_influence * 2 ]
			}
			resource_stockpile_compare = {
				resource = alloys
				value < @[ cultural_envoy_cost_alloys * 2 ]
			}
			resource_stockpile_compare = {
				resource = consumer_goods
				value < @[ cultural_envoy_cost_consumer_goods * 2 ]
			}
		}
	}
	max = 5
}
