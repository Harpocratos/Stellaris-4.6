# Fleet, starbase and defense platform policy helper effects. See common/policies/06_policies_fleets_and_starbases.txt

# Reinforced Plating: field sits on the starbase fleet so it covers its platforms. Scope: country
defense_platform_apply_reinforced_field = {
	every_owned_fleet = {
		limit = {
			is_defense_platform_fleet = yes
			NOT = { has_modifier = defense_platform_reinforced_field }
		}
		add_modifier = {
			modifier = defense_platform_reinforced_field
		}
	}
}

defense_platform_remove_reinforced_field = {
	every_owned_fleet = {
		limit = {
			has_modifier = defense_platform_reinforced_field
		}
		remove_modifier = defense_platform_reinforced_field
	}
}

# Resilient Frames: platforms disable near death instead of being destroyed. Scope: country
defense_platform_enable_resilience = {
	every_owned_fleet = {
		limit = { is_defense_platform_fleet = yes }
		every_owned_ship = {
			limit = {
				is_defense_platform_ship = yes
			}
			set_disable_at_health = 0.05
		}
	}
}

defense_platform_disable_resilience = {
	every_owned_fleet = {
		limit = { is_defense_platform_fleet = yes }
		every_owned_ship = {
			limit = {
				is_defense_platform_ship = yes
			}
			set_disable_at_health = no
			if = {
				limit = { is_disabled = yes }
				repair_percentage = 0.45
				set_disabled = no
			}
		}
	}
}

# Recover disabled platforms across the whole battle system. Scope: country, FROMFROM = the combat fleet
defense_platform_reenable_after_combat = {
	if = {
		limit = { exists = fromfrom } # guard against a destroyed fleet leaving a stale saved system
		fromfrom = { solar_system = { save_event_target_as = dp_combat_system } }
		every_owned_fleet = {
			limit = {
				is_defense_platform_fleet = yes
				solar_system = { is_same_value = event_target:dp_combat_system }
			}
			every_owned_ship = {
				limit = {
					is_defense_platform_ship = yes
					is_disabled = yes
				}
				repair_percentage = 0.45
				set_disabled = no
			}
		}
	}
}
