	set_country_code_flags = { is_nomadic = $IS_NOMAD$ }

# Removes every modifier the Grand Refurbishment situation (situation_grand_refurbishment) can apply to an arkship.
# Called both from the situation's on_abort and from on_colony_transfer (via arkship.1450), so a
# new owner is never left stuck with the situation's modifiers after the situation itself is gone. on_abort alone is
# not enough: when the former owner is destroyed the situation is torn down without running on_abort in a valid scope.
# Only lists modifiers that can be live mid-situation; the completion rewards (arkship.1380) are never present on abort.
# This = arkship colony.
remove_grand_refurbishment_arkship_modifiers_effect = {
	remove_modifier = refurbishment_ongoing
	remove_modifier = refurbishment_low_morale
	remove_modifier = refurbishment_low_morale_2
	remove_modifier = refurbishment_low_morale_3
	remove_modifier = refurbishment_allowances_granted
	remove_modifier = refurbishment_construction_fatigue
	remove_modifier = refurbishment_construction_fatigue_2
	remove_modifier = refurbishment_power_restoration
	remove_modifier = refurbishment_extended_displacement
	remove_modifier = refurbishment_administrative_burden
	remove_modifier = refurbishment_administrative_burden_gestalt
	remove_modifier = refurbishment_rework_zone
	remove_modifier = refurbishment_inspection_hold
	remove_modifier = refurbishment_bypass_routing
}

# Removes every modifier the nomad onboarding checklist (nomad_checklist_chain) can apply to the capital arkship.
# The checklist applies these in capital_scope (i.e. on the arkship colony, not the country), so they transfer to a
# conqueror who takes the arkship. Called from arkship.1450 on colony transfer so the new owner is not left with them.
# Covers both the in-progress penalties and the temporary completion rewards. This = arkship colony.
remove_checklist_arkship_modifiers_effect = {
	remove_modifier = checklist_unaligned_shields
	remove_modifier = checklist_aligned_shields
	remove_modifier = checklist_sensor_calibration
	remove_modifier = checklist_calibrated_sensors
	remove_modifier = checklist_corrosive_coolant
	remove_modifier = checklist_flushed_coolant
	remove_modifier = checklist_replicating_signal
	remove_modifier = checklist_deciphered_signal
	remove_modifier = checklist_harvest_resources_bonus
}

# Ends an in-progress Festival: stops the Celebration approach, starts the 5-year cooldown, and returns any encamped arkships to mobile.
# Guarded by the celebration_active flag so it is safe to call from multiple end paths (timer, manual un-encamp, Conserve approach).
# This = country
end_nomad_festival_effect = {
	if = {
		limit = {
			has_country_flag = celebration_active
		}
		remove_country_flag = celebration_active
		set_timed_country_flag = {
			flag = festival_cooldown
			days = @festival_cooldown_days_1800
		}
		random_situation = {
			limit = {
				is_situation_type = situation_nomad_economy
			}
			# Only revert the approach if the Festival is still the active one (the player may have switched to Conserve, etc.).
			if = {
				limit = {
					current_situation_approach = situation_nomad_economy_harvest_everything
				}
				set_situation_approach = situation_nomad_economy_smooth_spending
			}
			clear_variable = celebration_reserves_at_start
			clear_variable = celebration_months_remaining
		}
		every_owned_ship = {
			limit = {
				is_arkship_ship = yes
				is_system_locked = yes
			}
			# Clear the Festival tent ambient objects spread across the system's planets,
			# otherwise they linger until their duration expires after the arkship has left.
			solar_system = {
				every_system_ambient_object = {
					limit = {
						is_ambient_object_type = tent_ship_object
					}
					destroy_ambient_object = this
				}
			}
			set_system_locked = no
		}
	}
}

# Called from a Nomad-conquest vassal scope inside nomads.1 / nomads.5, after
# save_event_target_as = nomad_vassal and give_planet_and_system_to_rebel have
# run. For every war the overlord is in where fromfrom is also a participant,
# the new vassal (event_target:nomad_vassal) joins the war on the overlord's
# side. Determines the side explicitly so the engine doesn't reassign sides
# when a fresh subject is added.
nomad_vassal_join_overlord_wars_effect = {
	event_target:nomad_vassal = {
		random_agreement = {
			if = {
				limit = {
					NOT = {
						agreement_preset = preset_nomadic_purger
					}
				}
				set_agreement_preset = {
					preset = preset_nomad_tributary
				}
			}
			set_agreement_terms = {
				joins_overlord_wars = joins_overlord_wars_all
			}
		}
	}
	overlord = {
		every_war = {
			limit = {
				any_war_participant = {
					is_same_value = fromfrom
				}
				NOT = {
					any_war_participant = {
						is_same_value = event_target:nomad_vassal
					}
				}
			}
			save_event_target_as = nomad_war
			event_target:nomad_vassal = {
				if = {
					limit = {
						overlord = {
							is_war_participant = {
								war = event_target:nomad_war
								side = attackers
							}
						}
					}
					join_war_on_side = {
						war = event_target:nomad_war
						side = attackers
					}
				}
				else = {
					join_war_on_side = {
						war = event_target:nomad_war
						side = defenders
					}
				}
			}
		}
	}
}

give_planet_and_system_to_rebel = {
	$PLANET$ = {
		set_owner = $REBEL$
		solar_system = {
			if = {
				limit = { exists = starbase }
				starbase = {
					if = {
						limit = {
							is_owned_by = $OLD_OWNER$
						}
						set_owner = $REBEL$
						if = {
							limit = { has_starbase_size = starbase_outpost }
							set_starbase_size = starbase_starport
							set_starbase_module = {
								slot = 1
								module = shipyard
							}
							set_starbase_module = {
								slot = 2
								module = shipyard
							}
						}
						else_if = {
							limit = {
								NOT = { has_starbase_module = shipyard }
							}
							set_starbase_module = {
								slot = 1
								module = shipyard
							}
							set_starbase_module = {
								slot = 2
								module = shipyard
							}
						}
					}
				}
			}
			else = {
				$REBEL$ = {
					give_technology = { tech = tech_starbase_2 message = no }
					refresh_auto_generated_ship_designs = yes
				}
				create_starbase = {
					size = starbase_starport
					module = shipyard
					module = shipyard
					owner = $REBEL$
				}
			}
			add_claims = {
				who = $OLD_OWNER$
				show_notification = no
			}
			every_system_planet = {
				limit = {
					is_colony = yes
					NOT = { is_same_value = $PLANET$ }
					is_owned_by = $OLD_OWNER$
				}
				set_owner = $REBEL$
			}
		}
	}
}

champions_forge_pathfinding = {
	save_event_target_as = pathfinding_origin
	random_system = {
		limit = {
			NOR = {
				has_star_flag = guardian
				has_star_flag = hostile_system
			}
			OR = {
				NOT = { exists = space_owner }
				space_owner = { # We avoid evil systems. Makes for bad TV
					will_champions_forge_visit = yes
				}
			}
		}
		weights = {
			base = 1
			modifier = {
				add = 5
				is_capital_system = yes
			}
			modifier = {
				add = 5
				any_starbase_in_system = {
					has_starbase_size >= starbase_waystation_2
					owner? = {
						will_champions_forge_visit = yes
					}
				}
			}
			modifier = { # We are okay with short jumps
				add = 2
				distance = {
					source = prev
					type = hyperlane
					bypass_empire = prev.owner
					min_jumps = 1
					max_jumps = 5
				}
			}
			modifier = { # We prefer medium jumps
				add = 5
				distance = {
					source = prev
					type = hyperlane
					bypass_empire = prev.owner
					min_jumps = 6
					max_jumps = 10
				}
			}
			modifier = { # Long jumps are okay
				add = 2
				distance = {
					source = prev
					type = hyperlane
					bypass_empire = prev.owner
					min_jumps = 11
					max_jumps = 15
				}
			}
			modifier = { # Super Long jumps are bad
				add = 0.25
				distance = {
					source = prev
					type = hyperlane
					bypass_empire = prev.owner
					min_jumps = 16
					max_jumps = 1000
				}
			}
			modifier = {
				factor = 0.01
				OR = {
					space_owner? = {
						has_country_flag = champions_forge_recent_visit
					}
					any_starbase_in_system = {
						has_starbase_size >= starbase_waystation_2
						owner? = {
							has_country_flag = champions_forge_recent_visit
						}
					}
				}
			}
			modifier = {
				factor = 0.1
				NOR = {
					space_owner? = {
						has_communications = event_target:pathfinding_origin.owner
					}
					any_starbase_in_system = {
						has_starbase_size >= starbase_waystation_2
						owner? = {
							has_communications = event_target:pathfinding_origin.owner
						}
					}
				}
			}
		}
		random_system_planet = {
			limit = {
				is_colonizable = no
			}
			save_event_target_as = champions_forge_target
		}
		prev = {
			auto_move_to_planet = {
				target = event_target:champions_forge_target
				clear_auto_move_on_arrival = no
				arrival_effect = champions_forge_arrival_effect
			}
		}
	}
}

	orbit = {
		while = {
			count = 10
			random_list = {
				1 = {
					create_ambient_object = {
						type = caravaneer_billboard_1
						use_3d_location = yes
						entity_offset_height = { min = 5 max = 25 }
						entity_offset = { min = 10 max = 50 }
						entity_offset_angle = { min = 0 max = 350 }
						location = prev
					}
				}
				1 = {
					create_ambient_object = {
						type = caravaneer_billboard_2
						use_3d_location = yes
						entity_offset_height = { min = 5 max = 25 }
						entity_offset = { min = 10 max = 50 }
						entity_offset_angle = { min = 0 max = 350 }
						location = prev
					}
				}
				1 = {
					create_ambient_object = {
						type = caravaneer_billboard_3
						use_3d_location = yes
						entity_offset_height = { min = 5 max = 25 }
						entity_offset = { min = 10 max = 50 }
						entity_offset_angle = { min = 0 max = 350 }
						location = prev
					}
				}
			}
			last_created_ambient_object = {

			}
		}
	}
	if = {
		limit = {
			NOT = {

			set_country_flag = had_first_tournament

		}
	}
	else = {
		queue_actions = {
			wait = {
				duration = 180
			}
			effect = champions_forge_departure_effect
		}
	}

		every_system_ambient_object = {
			limit = {
				has_ambient_object_flag = champions_forge_arrive_object
			}

		}
		space_owner? = {
			set_timed_country_flag = {
				flag = champions_forge_recent_visit

			}
		}
		if = {
			limit = {

					set_timed_country_flag = {
						flag = champions_forge_recent_visit
						years = 20
					}
				}
			}
		}
	}
	champions_forge_pathfinding = yes
}

# Fallback effect if no winner is declared, cleaning up countries and removing disabled effects.
champions_forge_tournament_cleanup_effect = {
	every_country = {
		limit = {
			is_country_type = champions_forge_contestant
		}

		# Return the leased player fleets this vassal controls to their owners
		every_controlled_fleet = {
			limit = {
				has_fleet_flag = player_champions_forge_contestant_fleet
			}
			remove_fleet_flag = champions_forge_contestant_fleet
			remove_fleet_flag = player_champions_forge_contestant_fleet
			owner = {
				if = {
					limit = {
						any_owned_leader = {
							has_leader_flag = temp_contestant_admiral
						}
					}
					random_owned_leader = {
						limit = { has_leader_flag = temp_contestant_admiral }
						kill_leader = { show_notification = no }
					}
				}
			}
			# Undo the tournament's disable-at-health rigging and re-enable any ships
			# that were knocked out, wherever the fleet ended up.
			every_owned_ship = {
				set_disabled = no
				set_disable_at_health = no
			}
			end_fleet_contract = {
				initiator = this.controller
				reason = cancelled
			}
			clear_orders = yes
			set_mia = mia_return_home
			set_mia_return_delay = 180
		}

		# Destroy this vassal's own (AI) contestant fleets
		every_controlled_fleet = {
			limit = {
				has_fleet_flag = champions_forge_contestant_fleet
				NOT = { has_fleet_flag = player_champions_forge_contestant_fleet }
			}
			destroy_fleet = this
		}

		# Remove the temporary team country
		destroy_country = yes
	}

	# Clear the flag for ongoing Champions Forge tournaments
	event_target:champions_forge_enclave_country = {
		remove_country_flag = champions_forge_tournament_ongoing
		remove_country_flag = champions_forge_battle_started
	}

	# Send the enclave back out into the galaxy
	event_target:champions_forge_enclave_fleet = {
		set_timed_fleet_flag = {
			flag = champions_forge_match_cooldown
			years = 20
		}
		champions_forge_departure_effect = yes
	}
}

# Called from a country scope, the empire that is sending the contestant fleet
create_champions_forge_contestant = {
	save_event_target_as = champions_forge_contestant
	create_country = {
		name = {
			key = NAME_Contestant_Fleet
			variable_string = "\\[event_target:champions_forge_contestant.GetName\\]"
		}
		contact_rule = always
		ignore_initial_colony_error = yes
		type = champions_forge_contestant
		flag = event_target:champions_forge_contestant
		species = event_target:champions_forge_contestant
		effect = {
			save_event_target_as = champions_forge_contestant_team
			every_country = {
				limit = {
					NOR = {
						is_country_type = default
						is_same_value = prev
					}
				}
				set_faction_hostility = { # No more fighting amoeba
					target = prev
					set_hostile = no
					set_neutral = yes
					set_friendly = no
				}
			}
			if = {
				limit = {
					event_target:champions_forge_contestant_fleet = {
						exists = leader
					}
				}
				event_target:champions_forge_contestant_fleet.leader = {
					save_event_target_as = contestant_admiral
				}
			}
			else = {
				create_leader = {

# Increase the country variable Passenger Satisfaction Index for
# the Forever Cruise origin. It can never be 0, so a special check
# avoids this. The variable is used to set the monthly change
# of the situation.
increase_passenger_satisfaction = {
	hidden_effect = {
		random_situation = {
			limit = {
				is_situation_type = situation_passenger_satisfaction
			}
			if = {
				limit = {
					check_variable_arithmetic = {
						which = passenger_satisfaction_index
						value = @cruise_contract_sub
					}
				}
				set_variable = {
					which = passenger_satisfaction_index
					value = @cruise_contract_add
				}
			}
			else = {
				change_variable = {
					which = passenger_satisfaction_index
					value = @cruise_contract_add
				}
			}
			if = {
				limit = {
					situation_monthly_progress <= 0
				}
				set_timed_situation_flag = {
					flag = satisfaction_passengers_grace_period
					months = 24
				}
			}
			if = {
				limit = {
					OR = {
						current_stage = situation_passenger_satisfaction_stage_incensed
						current_stage = situation_passenger_satisfaction_stage_bored
					}
				}
				add_situation_progress = owner.value:passenger_satisfaction_passenger_contract
			}
		}
	}
}

## The Dyson Gun converts your entire energy stockpile into a shot, dealing 100th, 10th, full damage per level of the Dyson Gun
# THIS = Target System
# FROM = Megastructure
dyson_gun_firing = {
	from.owner = {
		export_resource_stockpile_to_variable = {
			resource = energy
			variable = dyson_gun_charge
		}
		set_resource = {
			resource = energy
			value = 0
		}
	}

	if = {
		# The Elder Voidspawn shrugs off the blast, then surges through space for two years
		limit = {
			any_ship_in_system = { is_ship_size = leviathan_01_voidspawn }
		}
		every_fleet_in_system = {
			limit = {
				any_owned_ship = { is_ship_size = leviathan_01_voidspawn }
			}
			add_modifier = {
				modifier = voidspawn_dyson_surge
				days = 730
			}
		}
	}
	else_if = {
		# A Contingency Sterilization Hub scatters the shot to a random system
		limit = {
			any_system_planet = {
				is_planet_class = pc_ai
				owner = { is_country_type = ai_empire }
			}
		}
		random_system = {
			limit = {
				NOT = { is_same_value = root }
			}
			dyson_gun_apply_damage = yes
		}
		# The Contingency reports that it has hijacked our targeting (FROM = the Contingency)
		random_system_planet = {
			limit = {
				is_planet_class = pc_ai
				owner = { is_country_type = ai_empire }
			}
			owner = { from.owner = { country_event = { id = nomads.8050 } } }
		}
	}
	else_if = {
		# Striking an Unbidden Dimensional Anchor only tears open a new rift, dealing no damage
		limit = {
			any_ship_in_system = { is_ship_size = dimensional_portal_ed }
		}
		create_unbidden_fleet = yes
		# The Unbidden savor the energy we just fed them (FROM = the Unbidden anchor's owner)
		random_ship_in_system = {
			limit = { is_ship_size = dimensional_portal_ed }
			owner = { from.owner = { country_event = { id = nomads.8055 } } }
		}
	}
	else_if = {
		# Once struck, Cetana reflects every following shot back at the attacker's capital
		limit = {
			any_ship_in_system = { is_ship_size = synth_queen_titan }
			from.owner = { has_country_flag = dyson_gun_struck_cetana }
		}
		from.owner = {
			capital_scope = {
				solar_system = {
					dyson_gun_apply_damage = yes
				}
			}
		}
		# Cetana reproaches us as she turns the shot back on our capital (FROM = Cetana)
		event_target:synth_queen_country_global = {
			from.owner = { country_event = { id = nomads.8065 } }
		}
	}
	else = {
		dyson_gun_apply_damage = yes
		# The first strike on Cetana primes her to reflect every shot to come
		if = {
			limit = {
				any_ship_in_system = { is_ship_size = synth_queen_titan }
			}
			from.owner = { set_country_flag = dyson_gun_struck_cetana }
			# Cetana voices her disappointment (FROM = Cetana)
			event_target:synth_queen_country_global = {
				from.owner = { country_event = { id = nomads.8060 } }
			}
		}
	}

	set_timed_star_flag = {
		flag = ai_dyson_gun_limit
		days = 3600
	}
}

## Devastates one colony from a Dyson Gun impact. A planetary shield generator blocks 90% of the devastation.
## Only fires while the shot cleared the energy threshold flag.
# THIS = Colony being hit
# FROM = Megastructure
# $AMOUNT$ = devastation on an unshielded colony, $SHIELDED$ = devastation behind a planetary shield generator
dyson_gun_devastate_colony = {
	if = {
		limit = {
			from.owner = { has_country_flag = dyson_gun_devastating_shot }
		}
		if = {
			limit = { has_planetary_shield_building = yes }
			add_planet_devastation = $SHIELDED$
		}
		else = {
			add_planet_devastation = $AMOUNT$
		}
	}
}

## Applies a Dyson Gun hit to one ship. Under the Synchronized Shielding doctrine, defense platforms and ion cannons take 60% less damage.
## This reduction is exclusive to the Stellar Cannon; normal combat is unaffected.
# THIS = Ship being hit
# FROM = Megastructure
dyson_gun_damage_ship = {
	if = {
		limit = {
			is_defense_platform_ship = yes
			# Synchronized Shielding doctrine shields platforms empire-wide
			owner = { has_policy_flag = defense_platform_synchronized_shielding }
		}
		damage_ship = {
			amount = from.owner.dyson_gun_damage_shielded
			attacker = from.owner
		}
	}
	else = {
		damage_ship = {
			amount = from.owner.dyson_gun_damage
			attacker = from.owner
		}
	}
}

## Resolves a Dyson Gun impact against the system it runs in: damages ships, devastates colonies, and raises the relevant diplomatic flags
# THIS = System being hit
# FROM = Megastructure
dyson_gun_apply_damage = {
	# Devastation is only inflicted when at least 10,000 energy was spent charging the shot
	from.owner = {
		if = {
			limit = {
				check_variable = {
					which = dyson_gun_charge
					value >= 10000
				}
			}
			set_country_flag = dyson_gun_devastating_shot
		}
		else = {
			remove_country_flag = dyson_gun_devastating_shot
		}
	}

	if = {
		limit = {
			from = {
				is_megastructure_type = dyson_gun_1
			}
		}
		from.owner = {
			set_variable = {
				which = dyson_gun_damage
				value = dyson_gun_charge
			}
			divide_variable = {
				which = dyson_gun_damage
				value = 100
			}
			set_variable = {
				which = dyson_gun_damage_shielded
				value = dyson_gun_damage
			}
			multiply_variable = {
				which = dyson_gun_damage_shielded
				value = 0.4
			}
		}
		every_ship_in_system = {
			dyson_gun_damage_ship = yes
		}
		every_system_planet = {
			limit = {
				is_colony = yes
			}
			dyson_gun_devastate_colony = {
				AMOUNT = 10
				SHIELDED = 1
			}
		}
	}
	if = {
		limit = {
			from = {
				is_megastructure_type = dyson_gun_2
			}
		}
		from.owner = {
			set_variable = {
				which = dyson_gun_damage
				value = dyson_gun_charge
			}
			divide_variable = {
				which = dyson_gun_damage
				value = 50
			}
			set_variable = {
				which = dyson_gun_damage_shielded
				value = dyson_gun_damage
			}
			multiply_variable = {
				which = dyson_gun_damage_shielded
				value = 0.4
			}
		}
		every_ship_in_system = {
			dyson_gun_damage_ship = yes
		}
		every_system_planet = {
			limit = {
				is_colony = yes
			}
			dyson_gun_devastate_colony = {
				AMOUNT = 25
				SHIELDED = 2.5
			}
		}
	}
	if = {
		limit = {
			from = {
				is_megastructure_type = dyson_gun_3
			}
		}
		from.owner = {
			set_variable = {
				which = dyson_gun_damage
				value = dyson_gun_charge
			}
			divide_variable = {
				which = dyson_gun_damage
				value = 10
			}
			set_variable = {
				which = dyson_gun_damage_shielded
				value = dyson_gun_damage
			}
			multiply_variable = {
				which = dyson_gun_damage_shielded
				value = 0.4
			}
		}

		every_ship_in_system = {
			dyson_gun_damage_ship = yes
		}
		every_system_planet = {
			limit = {
				is_colony = yes
			}
			dyson_gun_devastate_colony = {
				AMOUNT = 50
				SHIELDED = 5
			}
		}
		every_system_planet = {
			limit = {
				from.owner = {
					check_variable = {
						which = dyson_gun_charge
						value > 50000
					}
				}
				# A planetary shield generator blocks the terraforming heat
				has_planetary_shield_building = no
			}
			increase_planet_temperature_without_habitables = yes
		}
		if = {
			limit = {
				OR = {
					is_active_resolution = "resolution_ecology_environmental_control_board"
					is_active_resolution = "resolution_ecology_paradise_initiative"
				}
				from.owner = {
					check_variable = {
						which = dyson_gun_charge
						value > 50000
					}
				}
			}
			from.owner = {
				set_timed_country_flag = { flag = resolution_breached_terraformed days = 3600 }

		}

	if = {

heirs_khan_dummy_country_system_finder = {
	if = {
		limit = {
			any_owned_colony = {
				solar_system? = {
					NOT = { exists = space_owner }
				}
			}
		}
		random_owned_colony = {
			limit = {
				solar_system? = {
					NOT = { exists = space_owner }
				}
			}
			weights = {
				base = 1
				modifier = {
					factor = 100
					is_capital = yes
				}
			}
			solar_system = { make_heirs_khan_dummy_country = { PLAYER_COUNTRY = root } }
		}
	}
	else_if = {
		limit = {
			any_owned_nonprimary_starbase = {
				is_waystation_starbase = yes
				solar_system? = {
					NOT = { exists = space_owner }
				}
			}
		}
		random_owned_nonprimary_starbase = {
			limit = {
				is_waystation_starbase = yes
				solar_system? = {
					NOT = { exists = space_owner }
				}
			}
			solar_system = { make_heirs_khan_dummy_country = { PLAYER_COUNTRY = root } }
		}
	}
	else = {
		random_system = {
			limit = {
				count_system_planet_colony = {
					count = 0
				}
				count_system_megastructure = {
					count = 0
				}
			}
			weights = {
				base = 1
				modifier = {
					factor = 100
					NOT = { exists = space_owner }
				}
				modifier = {
					factor = 10
					root = {
						any_owned_colony = {
							solar_system? = {
								is_same_value = prevprevprev
							}
						}
					}
				}
				modifier = {
					factor = 5
					root = {
						any_owned_nonprimary_starbase = {
							is_waystation_starbase = yes
							solar_system? = {
								is_same_value = prevprevprev
							}
						}
					}
				}
			}
			make_system_ownerless_effect = yes
			make_heirs_khan_dummy_country = { PLAYER_COUNTRY = root }
		}
	}
}

# Nomad Raid War Effects
# Transfers the connected wayline of event_target:nomad_raid_defender that borders
# the winner's waystation network. Flood-fills along the chain from the border point.
# this = attacker country
nomad_raid_loot_waystations_effect = {
	save_event_target_as = nomad_raider
	event_target:nomad_raid_defender = {
		# Step 1: Flag defender waystation systems directly bordering the attacker's network
		every_controlled_fleet = {
			limit = {
				is_waystation_ship = yes
				solar_system? = {
					any_neighbor_system = {
						any_fleet_in_system = {
							is_waystation_ship = yes
							owner = { is_same_value = event_target:nomad_raider }
						}
					}
				}
			}
			solar_system = { set_star_flag = nomad_raid_contested_waystation }
		}
		# Step 2: Flood-fill along the connected wayline
		while = {
			limit = {
				any_controlled_fleet = {
					is_waystation_ship = yes
					NOT = { solar_system? = { has_star_flag = nomad_raid_contested_waystation } }
					solar_system? = {
						any_neighbor_system = { has_star_flag = nomad_raid_contested_waystation }
					}
				}
			}
			every_controlled_fleet = {
				limit = {
					is_waystation_ship = yes
					NOT = { solar_system? = { has_star_flag = nomad_raid_contested_waystation } }
					solar_system? = {
						any_neighbor_system = { has_star_flag = nomad_raid_contested_waystation }
					}
				}
				solar_system = { set_star_flag = nomad_raid_contested_waystation }
			}
		}
		# Step 3: Transfer all flagged waystations to the attacker
		every_controlled_fleet = {
			limit = {
				is_waystation_ship = yes
				solar_system? = { has_star_flag = nomad_raid_contested_waystation }
			}
			set_owner = event_target:nomad_raider
		}
	}
	# Step 4: Clean up flags (transferred waystations now belong to the attacker)
	every_controlled_fleet = {
		limit = {
			is_waystation_ship = yes
			solar_system? = { has_star_flag = nomad_raid_contested_waystation }
		}
		solar_system = { remove_star_flag = nomad_raid_contested_waystation }
	}
}

# Absorbs a fraction of a doomed planet's population onto the conqueror's capital arkship, mirroring how a settled empire retains conquered pops.

}

				}
			}
		}
		every_fleet_in_system = {
			limit = {
				is_waystation_ship = yes
				owner? = {
					is_same_value = event_target:defender
				}
			}
			destroy_fleet = this
		}
	}
	# Safety net (lowest priority): an unclaimed bordering system.
	random_system_within_border = {
		limit = {
			is_on_border = yes
			any_neighbor_system = { has_owner = no }
		}
		random_neighbor_system = {
			limit = { has_owner = no }
			save_event_target_as = expulsion_system
		}
	}
	# Preferred: a bordering system whose owner still admits the expelled empire. Overrides the safety net when one exists.
	# Exclude owners at war with us: they grant fleet access by default, but the impending truce would trap the fleets there.
	random_system_within_border = {
		limit = {
			is_on_border = yes
			any_neighbor_system = {
				has_owner = yes
				has_access_fleet = event_target:defender
				owner = { NOT = { is_at_war_with = event_target:defender } }
			}
		}
		random_neighbor_system = {
			limit = {
				has_owner = yes
				has_access_fleet = event_target:defender
				owner = { NOT = { is_at_war_with = event_target:defender } }
			}
			save_event_target_as = expulsion_system
		}
	}
	every_system_within_border = {
		limit = {
			any_fleet_in_system = {
				owner? = {
					is_same_value = event_target:defender
				}
				is_expellable_nomad_fleet = yes
			}
		}
		every_fleet_in_system = {
			limit = {
				owner? = {
					is_same_value = event_target:defender
				}
				is_expellable_nomad_fleet = yes
			}
			# Stop the fleet first so the relocation settles it into orbit instead of letting it resume its old course.
			clear_orders = yes

expel_nomads_destroy_waystation_and_take_stockpile_effect = {
	save_event_target_as = system_owner
	every_system_within_border = {
		limit = {
			any_fleet_in_system = {
				is_waystation_ship = yes
				owner? = {
					is_same_value = event_target:defender
				}
			}
		}
		every_fleet_in_system = {
			limit = {
				is_waystation_ship = yes
				owner? = {
					is_same_value = event_target:defender
				}
			}
			starbase? = {
				ship? = {
					transfer_resource_stockpile = {
						target = event_target:system_owner
					}
				}
			}
			destroy_fleet = this
		}
	}
	# Safety net (lowest priority): an unclaimed bordering system.
	random_system_within_border = {
		limit = {
			is_on_border = yes
			any_neighbor_system = { has_owner = no }
		}
		random_neighbor_system = {
			limit = { has_owner = no }
			save_event_target_as = expulsion_system
			set_star_flag = expulsion_system
		}
	}
	# Preferred: a bordering system whose owner still admits the expelled empire. Overrides the safety net when one exists.
	# Exclude owners at war with us: they grant fleet access by default, but the impending truce would trap the fleets there.
	random_system_within_border = {
		limit = {
			is_on_border = yes
			any_neighbor_system = {
				has_owner = yes
				has_access_fleet = event_target:defender
				owner = { NOT = { is_at_war_with = event_target:defender } }
			}
		}
		random_neighbor_system = {
			limit = {
				has_owner = yes
				has_access_fleet = event_target:defender
				owner = { NOT = { is_at_war_with = event_target:defender } }
			}
			save_event_target_as = expulsion_system
			set_star_flag = expulsion_system
		}
	}
	every_system_within_border = {
		limit = {
			any_fleet_in_system = {
				owner? = {
					is_same_value = event_target:defender
				}
				is_expellable_nomad_fleet = yes
			}
		}
		every_fleet_in_system = {
			limit = {
				owner? = {
					is_same_value = event_target:defender
				}
				is_expellable_nomad_fleet = yes
			}
			# Stop the fleet first so the relocation settles it into orbit instead of letting it resume its old course.
			clear_orders = yes
			set_location = event_target:expulsion_system
		}
	}
	event_target:defender = {
		set_timed_country_flag = {
			flag = expelled_from_@event_target:system_owner
			years = 10 		# This should match the TRUCE_YEARS define
		}
	}
}

		DURATION = $DURATION$
		TARGET = $TARGET$
		EFFECT = $EFFECT$

	# and impose a penalty to mining stations in orbit of the planet.
	add_modifier = {
		modifier = harvested_resources_mining
		months = $DURATION$
		time_multiplier = from.controller.value:arkship_harvest_resources_duration_calc
	}
	if = {
		limit = {
			from.controller = {
				can_access_shroud = yes

				NOT = {
					any_system_planet = {

espionage_support_effect = {
	random_spynetwork = {
		limit = {
			has_spynetwork_flag = spy_support_contract_network@root
		}
		remove_spynetwork_flag = spy_support_contract_network@root
		if = {
			limit = {
				exists = root
			}
			hidden_effect = {
				random_list = { #sometimes you get small bonus, sometimes bigger bonus
					7 = {
						add_modifier = {
							modifier = nomad_espionage_support
							days = 1200
						}
					}
					1 = {
						modifier = {
							add = root.espionage_support_counter
						}
						add_modifier = {
							modifier = nomad_espionage_support
							days = 1200
							mult = 2
						}
					}
					1 = {
						modifier = {
							factor = root.espionage_support_counter
						}
						add_modifier = {
							modifier = nomad_espionage_support
							days = 1200
							mult = root.espionage_support_counter
						}
					}
				}
				root = { clear_variable = espionage_support_counter }
			}
		}
	}
	add_intel = {
		amount = $INTEL_AMOUNT$
		who = fromfrom.space_owner
	}
}

create_princess_poi = {
	random_system = {
		limit = {
			NOR = {
				has_star_flag = sealed_system
				has_star_flag = hostile_system
				has_star_flag = princess_poi_system
			}
			distance = {
				source = root.capital_scope.solar_system
				min_jumps = 2
			}
			any_system_planet = {
				has_any_flag = no
				OR = {
					is_planet_class = pc_asteroid
					is_planet_class = pc_gas_giant
					is_planet_class = pc_barren
				}
			}
		}
		set_star_flag = princess_poi_system
		random_system_planet = {
			limit = {
				has_any_flag = no
				OR = {
					is_planet_class = pc_asteroid
					is_planet_class = pc_gas_giant
					is_planet_class = pc_barren
				}
			}
			set_planet_flag = princess_poi_planet_$NUMBER$
			save_event_target_as = nomadic_princess_poi_target_$NUMBER$
			create_ambient_object = {
				type = small_debris_object
			}
			last_created_ambient_object = {
				set_location = {
					target = event_target:nomadic_princess_poi_target_$NUMBER$
					distance = 10
					angle = 180
				}
			}
		}
	}
	create_point_of_interest = {
		id = nomadic_princess_poi_$NUMBER$
		name = nomadic_princess_poi_title_$NUMBER$
		desc = nomadic_princess_poi_desc_$NUMBER$
		event_chain = nomadic_princess_chain
		location = event_target:nomadic_princess_poi_target_$NUMBER$
	}
}

	set_country_flag = checklist_decipher_signal
	closest_system = {
		min_steps = 5
		limit = {
			OR = {
				# Is the capital system of a regular empire.
				AND = {
					is_capital_system = yes
					owner? = {
						is_candidate_for_early_contact = yes
						NOT = {
							has_communications = root
						}
					}
				}
				# Is an enclave system and the enclave hasn't been killed.
				AND = {
					has_star_flag = enclave
					any_fleet_in_system = {
						owner? = {
							OR = {
								is_same_value = event_target:artist_enclave_country
								is_same_value = event_target:curator_enclave_country
								has_country_flag = trader_enclave_country
								is_same_value = event_target:mindwarden_enclave_country
								is_same_value = event_target:salvager_enclave_country
								is_same_value = event_target:shroudwalker_enclave_country
								is_same_value = event_target:mindwarden_enclave_country
							}
							NOT = {
								has_communications = root
							}
						}
					}
				}
				# Is Chor's Compass
				AND = {
					has_star_flag = caravaneer_home
					any_fleet_in_system = {
						owner? = {
							is_same_value = event_target:caravaneer_home_country
							NOT = {
								has_communications = root
							}
						}
					}
				}
			}
		}

		random_fleet_in_system = {
			limit = {
				owner = {
					is_space_critter_country = no
				}

			owner = {
				save_global_event_target_as = comms_source@root
			}
		}
	}
	enable_mission = {
		name = NOMAD_CHECKLIST_DECIPHER_SIGNAL
		location = event_target:comms_signal_system@root

# Spawns a precursor special project on a valid uninhabitable planet in a precursor system.
# Non-nomads pick a system within border; Nomads search the own system and neighbors of a
# deployed waystation that is adjacent to the relevant precursor system.
# Usage: spawn_precursor_anomaly_effect = { PRECURSOR_FLAG = precursor_1 PROJECT_NAME = VULTAUM_ARTIFACT_FOUND }
spawn_precursor_anomaly_effect = {
	if = {
		limit = { is_nomadic = no }
		random_system_within_border = {
			limit = {
				has_star_flag = $PRECURSOR_FLAG$
				any_system_planet = {
					uninhabitable_regular_planet = yes
					NOT = { has_planet_flag = precursor_world }
					has_owner = no
					has_anomaly = no
					is_surveyed = {
						who = root
						status = yes
					}
				}
			}
			random_system_planet = {
				limit = {
					uninhabitable_regular_planet = yes
					NOT = { has_planet_flag = precursor_world }
					has_owner = no
					has_anomaly = no
					is_surveyed = {
						who = root
						status = yes
					}
				}
				save_event_target_as = precursor_planet
				set_planet_flag = precursor_world
				enable_special_project = {
					name = $PROJECT_NAME$
					location = this
					owner = root
				}
			}
		}
	}
	else = {
		# Nomad path: find a waystation whose own system or a neighbor contains the precursor system
		random_owned_nonprimary_starbase = {
			limit = {
				is_waystation_starbase = yes
				solar_system = {
					OR = {
						AND = {
							has_star_flag = $PRECURSOR_FLAG$
							any_system_planet = {
								uninhabitable_regular_planet = yes
								NOT = { has_planet_flag = precursor_world }
								has_owner = no
								has_anomaly = no
								is_surveyed = {
									who = root
									status = yes
								}
							}
						}
						any_neighbor_system = {
							has_star_flag = $PRECURSOR_FLAG$
							any_system_planet = {
								uninhabitable_regular_planet = yes
								NOT = { has_planet_flag = precursor_world }
								has_owner = no
								has_anomaly = no
								is_surveyed = {
									who = root
									status = yes
								}
							}
						}
					}
				}
			}
			solar_system = {
				if = {
					limit = {
						has_star_flag = $PRECURSOR_FLAG$
						any_system_planet = {
							uninhabitable_regular_planet = yes
							NOT = { has_planet_flag = precursor_world }
							has_owner = no
							has_anomaly = no
							is_surveyed = {
								who = root
								status = yes
							}
						}
					}
					random_system_planet = {
						limit = {
							uninhabitable_regular_planet = yes
							NOT = { has_planet_flag = precursor_world }
							has_owner = no
							has_anomaly = no
							is_surveyed = {
								who = root
								status = yes
							}
						}
						save_event_target_as = precursor_planet
						set_planet_flag = precursor_world
						enable_special_project = {
							name = $PROJECT_NAME$
							location = this
							owner = root
						}
					}
				}
				else = {
					random_neighbor_system = {
						limit = {
							has_star_flag = $PRECURSOR_FLAG$
							any_system_planet = {
								uninhabitable_regular_planet = yes
								NOT = { has_planet_flag = precursor_world }
								has_owner = no
								has_anomaly = no
								is_surveyed = {
									who = root
									status = yes
								}
							}
						}
						random_system_planet = {
							limit = {
								uninhabitable_regular_planet = yes
								NOT = { has_planet_flag = precursor_world }
								has_owner = no
								has_anomaly = no
								is_surveyed = {
									who = root
									status = yes
								}
							}
							save_event_target_as = precursor_planet
							set_planet_flag = precursor_world
							enable_special_project = {
								name = $PROJECT_NAME$
								location = this
								owner = root
							}
						}
					}
				}
			}
		}
	}
	set_country_flag = ongoing_precursor_artifact
}

# Selects precursor_spawn_system for use with spawn_system.
# Works for both regular empires (uses border systems) and nomads (uses waystations/colonies).
# Called from country scope.
precursor_capital_spawn_system_effect = {
	if = {
		limit = { is_nomadic = no }
		if = {
			limit = { any_system_within_border = { has_star_flag = $PRECURSOR_FLAG$ } }
			random_system_within_border = {
				limit = { has_star_flag = $PRECURSOR_FLAG$ }
				save_event_target_as = precursor_spawn_system
			}
		}
		else = {
			random_system_within_border = {
				save_event_target_as = precursor_spawn_system
			}
		}
	}
	else = {
		if = {
			limit = {
				any_owned_nonprimary_starbase = {
					is_waystation_starbase = yes
					solar_system? = { has_star_flag = $PRECURSOR_FLAG$ }
				}
			}
			random_owned_nonprimary_starbase = {
				limit = {
					is_waystation_starbase = yes
					solar_system? = { has_star_flag = $PRECURSOR_FLAG$ }
				}
				solar_system = { save_event_target_as = precursor_spawn_system }
			}
		}
		else_if = {
			limit = {
				any_owned_nonprimary_starbase = { is_waystation_starbase = yes }
			}
			random_owned_nonprimary_starbase = {
				limit = { is_waystation_starbase = yes }
				solar_system = { save_event_target_as = precursor_spawn_system }
			}
		}
		else = {
			random_owned_colony = {
				solar_system? = { save_event_target_as = precursor_spawn_system }
				weights = {
					base = 1
					modifier = {
						add = 10
						solar_system? = { has_star_flag = $PRECURSOR_FLAG$ }
					}
				}
			}
		}
	}
}

# Performs a population exchange between event_target:migrating_arkship and event_target:migrating_planet,
# applies the per-host-planet 5-year cooldown, and notifies the host empire.
# Callers must save migrating_planet and migrating_arkship before invoking.
arkship_migration_effect = {
	event_target:migrating_planet = {
		set_timed_carrier_flag = {
			flag = migration_happened_planet
			years = 5
		}
		owner = {
			create_message = {
				type = MESSAGE_NOMAD_MIGRATION_HOST
				localization = MESSAGE_NOMAD_MIGRATION_HOST_DESC
				days = 30
				target = root
				variable = {
					type = name
					localization = COUNTRY
					scope = root
				}
				variable = {
					type = name
					localization = PLANET
					scope = event_target:migrating_planet
				}
				variable = {
					type = name
					localization = ARKSHIP
					scope = event_target:migrating_arkship
				}
			}
		}
		random_owned_pop_group = {
			resettle_pop_group = {
				POP_GROUP = this
				PLANET = event_target:migrating_arkship
				AMOUNT = value:compare_resettle_attraction|ORIGIN|event_target:migrating_planet|DEST|event_target:migrating_arkship|
			}
		}
	}
	event_target:migrating_arkship = {
		random_owned_pop_group = {
			resettle_pop_group = {
				POP_GROUP = this
				PLANET = event_target:migrating_planet
				AMOUNT = value:compare_resettle_attraction|ORIGIN|event_target:migrating_arkship|DEST|event_target:migrating_planet|
			}
		}
	}
	# Forever Cruise: set new migrant species to passenger citizenship
	if = {
		limit = {
			event_target:migrating_arkship.owner = {
				has_origin = origin_forever_cruise
				NOT = { has_country_flag = forever_cruise_over }
				is_hospitality_servitor_empire = no
			}
		}
		event_target:migrating_arkship.owner = {
			every_owned_species = {
				limit = {
					NOR = {
						is_same_species = prev
						has_citizenship_type = { type = citizenship_cruise_crew country = prev }
						has_citizenship_type = { type = citizenship_cruise_passenger country = prev }
					}
				}
				set_citizenship_type = {
					type = citizenship_cruise_passenger
					country = prev
				}
				set_living_standard = {
					type = living_standard_cruise_passenger
					country = prev
				}
			}
		}
	}
	# Hospitality Servitor: set new migrant species to organic trophy
	else_if = {
		limit = {
			event_target:migrating_arkship.owner = {
				is_hospitality_servitor_empire = yes
			}
		}
		event_target:migrating_arkship.owner = {
			every_owned_species = {
				limit = {
					NOR = {
						is_same_species = prev
						has_citizenship_type = { type = citizenship_cruise_crew country = prev }
						has_citizenship_type = { type = citizenship_organic_trophy country = prev }
					}
				}
				set_citizenship_type = {
					type = citizenship_organic_trophy
					country = prev
				}
				set_living_standard = {
					type = living_standard_organic_trophy
					country = prev
				}
			}
		}
	}
}

# Releases ownership of the scoped system by destroying every starbase in it.
# Sends mobile fleets MIA and dismantles planetary orbital stations.
# Scope: galactic_object
# Optional params:
#   DONT_DISPLACE = <country target> # fleets owned by this country are left in place
make_system_ownerless_effect = {
	hidden_effect = {
		every_fleet_in_system = {
			limit = {
				is_mobile = yes
				exists = owner
				[[DONT_DISPLACE]
					NOT = { is_owned_by = $DONT_DISPLACE$ }
				]
			}
			set_mia = mia_return_home
		}
		every_system_planet = {
			limit = { exists = orbital_station }
			orbital_station = { dismantle = yes }
		}
		space_owner? = {
			every_owned_nonprimary_starbase = {
				limit = { solar_system = { is_same_value = prev.prev.prev } }
				ship = { destroy_fleet = this }
			}
		}
		every_starbase_in_system = {
			ship = { destroy_fleet = this }
		}
	}
}

# Scope: subject country. Erases and re-applies tribute_vault_loyalty based on tribute_vault_count.
refresh_tribute_vault_loyalty = {
	erase_modifier = {
		MOD = tribute_vault_loyalty
	}
	if = {
		limit = {
			is_variable_set = tribute_vault_count
			check_variable = {
				which = tribute_vault_count
				value > 0
			}
		}
		add_modifier = {
			modifier = tribute_vault_loyalty
			multiplier = value:tribute_vault_count_mult
			days = -1
		}
	}
}
