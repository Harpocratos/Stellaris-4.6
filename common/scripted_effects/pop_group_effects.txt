pop_amount_change_category = {
	colony = {
		create_pop_group = {
			pop_group = root
			category = $CATEGORY$
			size = 0
			effect = {
				transfer_pop_amount = {
					source = root
					target = this
					amount = $AMOUNT$
					growth_category = "GROWTH_CAT_PROMOTION"
				}
			}
		}
	}
}
