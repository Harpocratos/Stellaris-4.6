# General term for ideologies based on a specific ideology (if existent)
# Scopes = country, ideology
defined_text = {
	name = GetSpecificIdeologyType
	text = {
		trigger = {
			OR = {
				AND = {
					is_scope_type = country
					OR = {
						ideology? = { is_spiritualist = yes }
						AND = {
							NOT = { exists = ideology }
							is_spiritualist = yes
						}
					}
				}
				AND = {
					is_scope_type = ideology
					is_spiritualist = yes
				}
			}
		}
		localization_key = IDEOLOGY_TYPE_SPIRITUALIST
	}
	text = {
		trigger = {
			AND = {
				is_scope_type = country
				is_hive_empire = yes
				OR = {
					ideology? = { is_spiritualist = no }
					NOT = { exists = ideology }
				}
			}
		}
		localization_key = IDEOLOGY_TYPE_HIVE
	}
	text = {
		trigger = {
			AND = {
				is_scope_type = country
				is_machine_empire = yes
				OR = {
					ideology? = { is_spiritualist = no }
					NOT = { exists = ideology }
				}
			}
		}
		localization_key = IDEOLOGY_TYPE_MACHINE
	}
	text = {
		trigger = {
			is_scope_type = ideology
			is_spiritualist = no
		}
		localization_key = IDEOLOGY_LOWER
	}
	default = IDEOLOGY_TYPE_DEFAULT
}
