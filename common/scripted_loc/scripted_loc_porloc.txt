defined_text = {
	name = GetPlanetMoonBR
	text = {
		trigger = { is_planet_class = pc_habitat }
		localization_key = PORLOC_habitat_lowercase
	}
	text = {
		trigger = {
			OR = {
				is_planet_class = pc_ringworld_habitable
				is_planet_class = pc_ringworld_habitable_damaged
				is_planet_class = pc_cybrex
			}
		}
		localization_key = PORLOC_ringworld_lowercase
	}
	text = {
		trigger = {
			is_planet_class = pc_ark
		}
		localization_key = PORLOC_arkship
	}
	text = {
		trigger = {
			is_moon = yes
		}
		localization_key = PORLOC_moon
	}
	text = {
		trigger = {
			is_artificial = no
			is_moon = no
		}
		localization_key = PORLOC_planet
	}
}
