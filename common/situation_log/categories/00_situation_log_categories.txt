# category = {
# 	collapsible = <yes/no>		# whether the entry can be collapsed, default is yes
#	icon = <GFX_icon>			# The icon used for the category header
# 	priority = <integer>		# numerical value that indicate how high in the list the entry should appear
								# The higher value, the lower in the list
								# In case of equality, the tie breaker will be an alphabetical sort of the

# }
