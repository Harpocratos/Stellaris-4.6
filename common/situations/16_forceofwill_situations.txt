situation_ideological_revolt = {
	picture = GFX_evt_open_revolt
	category = negative
	situation_log_category = urgent_matters
	revolt_situation = yes

	desc = {
		trigger = { has_situation_flag = schism_revolt }
		text = situation_ideological_revolt_desc_schism
	}
	desc = {
		trigger = {
			NOT = { has_situation_flag = schism_revolt }
			target.ideology? = {
				is_same_value = event_target:ideological_revolt_target_ideology@root
			}
		}
		text = situation_ideological_revolt_desc_martyr_drift_specific
	}
	desc = {
		trigger = {
			NOR = {
				has_situation_flag = schism_revolt
				target.ideology? = {
					is_same_value = event_target:ideological_revolt_target_ideology@root
				}
			}
		}
		text = situation_ideological_revolt_desc_martyr_drift
	}

	fail_icon = GFX_situation_outcome_meh
	fail_icon_frame = GFX_situation_outcome_frame_green
	complete_icon = GFX_situation_outcome_revolt
	complete_icon_frame = GFX_situation_outcome_frame_red

	on_start = {
		set_situation_flag = ideological_revolt_situation_flag
	}

	on_progress_complete = {
		random_country = {
			limit = {
				has_country_flag = revolt_joiner_empire_@root
				country_can_back_ideological_revolt_trigger = {
					SITUATION_OWNER = root.owner
				}
			}
			save_event_target_as = new_owner_empire
		}
		if = {
			limit = { exists = event_target:new_owner_empire }
			# Told from the backer's own scope, since a tooltip cannot read event targets.
			event_target:new_owner_empire = {
				if = {
					limit = {
						ideological_revolt_creates_vassal_trigger = {
							BACKER = this
							TARGET = prev.target
						}
					}
					custom_tooltip = situation_ideological_revolt_complete_joiner_nomadic_tooltip
				}
				else = {
					custom_tooltip = situation_ideological_revolt_complete_joiner_tooltip
				}
			}
		}
		else_if = {
			limit = {
				NOT = { has_situation_flag = schism_revolt }
				target.ideology? = {
					is_same_value = event_target:ideological_revolt_target_ideology@root
				}
			}
			custom_tooltip = situation_ideological_revolt_complete_specific_tooltip
		}
		else = { custom_tooltip = situation_ideological_revolt_complete_tooltip }
		hidden_effect = {
			target = {
				set_timed_planet_flag = {
					flag = recent_revolt_attempt
					days = 5400
				}
				remove_modifier = envoy_martyr_modifier
				remove_modifier = open_defiance
				if = {
					limit = { exists = event_target:envoy_martyr_ideology@this }
					clear_global_event_target = envoy_martyr_ideology@this
				}
			}
			owner = {
				set_timed_country_flag = {
					flag = had_recent_successful_rebellion
					years = 10
				}
				if = {
					limit = {
						NOT = { is_variable_set = accept_ideological_revendication }
					}
					set_variable = {
						which = accept_ideological_revendication
						value = 0
					}
				}
				set_variable_to_random_value = {
					which = accept_ideology_change
					min = 1
					max = 3
				}
				change_variable = {
					which = accept_ideological_revendication
					value = accept_ideology_change
				}
				clear_variable = accept_ideology_change
			}
			if = {
				limit = { exists = event_target:new_owner_empire }
				situation_event = { id = forceofwill.7335 }
			}
			else_if = {
				limit = { has_situation_flag = schism_revolt }
				situation_event = { id = forceofwill.7385 }
			}
			else = {
				situation_event = { id = forceofwill.7340 }
			}
		}
	}

	on_fail = {
		custom_tooltip = situation_ideological_revolt_fail_tooltip
		hidden_effect = {
			target = {
				set_timed_planet_flag = {
					flag = recent_revolt_attempt
					days = 5400
				}
				remove_modifier = envoy_martyr_modifier
				remove_modifier = open_defiance
				if = {
					limit = { exists = event_target:envoy_martyr_ideology@this }
					clear_global_event_target = envoy_martyr_ideology@this
				}
			}
			situation_event = { id = forceofwill.7345 }
			if = {
				limit = {
					owner? = { is_variable_set = accept_ideological_revendication }
					OR = {
						current_situation_approach = approach_situation_ideological_revolt_adopt_ideology
						current_situation_approach = approach_situation_ideological_revolt_abandon_ideology
					}
				}
				owner = { clear_variable = accept_ideological_revendication }
			}
		}
	}

	abort_trigger = {
		NOT = {
			owner = { #otherwise, those events flip ownership, which immediately invalidates the Situation
				OR = {
					has_active_event = { forceofwill.7335 }
					has_active_event = { forceofwill.7340 }
					has_active_event = { forceofwill.7385 }
				}
			}
		}
		OR = {
			NOT = { exists = target.owner }
			target.owner = { NOT = { is_same_value = root.owner } }
			target.controller = { NOT = { is_same_value = root.owner } }
			AND = {
				has_situation_flag = schism_revolt
				NOT = { exists = owner.ideology }
			}
			AND = {
				NOT = { has_situation_flag = schism_revolt }
				exists = owner.ideology

			}
		}
	}

				}
			}
			owner = { clear_variable = accept_ideological_revendication }
		}
		clear_global_event_target = ideological_revolt_target_ideology@this
		target? = {
			remove_modifier = envoy_martyr_modifier
			remove_modifier = open_defiance
			if = {
