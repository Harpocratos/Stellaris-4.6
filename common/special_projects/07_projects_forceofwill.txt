# Adopt Ideology / Shift Ethics
special_project = {
	key = "ADOPT_IDEOLOGY_SPECIAL_PROJECT"
	cost = {
		base = 0
		# ideology still has to be adopted
		modifier = {
			add = 5000
			has_country_flag = adopt_ideology_yourself
			desc = "ADOPT_IDEOLOGY"
		}
		# an ethic shift is needed
		modifier = {
			add = 2500
			has_country_flag = adopt_ideology_needs_ethic_shift
			has_country_flag = adopt_ideology_yourself
			desc = "SHIFT_ETHICS"
		}
	}

	days_to_research = 180
	tech_department = society_technology
	timelimit = -1
	picture = GFX_evt_ideological_council
	icon = "gfx/interface/icons/situation_log/situation_log_main_quest.dds"
	location = no
	sound = yes

	event_scope = country_event

	on_success = {
		if = {
			limit = {
				OR = {
					NOT = { exists = ideology }
					ideology = { NOT = { is_same_value = event_target:ideology_target } }
				}
			}
			if = {
				limit = { has_country_flag = adopting_ideology_from_invitation_@event_target:ideology_target }
				remove_country_flag = adopting_ideology_from_invitation_@event_target:ideology_target
				every_playable_country = {
					limit = {
						has_country_flag = invited_to_our_ideology_@prev
						ideology? = { is_same_value = event_target:ideology_target }
					}
					remove_country_flag = invited_to_our_ideology_@prev
					set_country_flag = achievement_invite_ideology
				}
			}
			adopt_ideology = event_target:ideology_target

			country_event = {
				id = forceofwill.1045
				scopes = { from = event_target:ideology_target }
			}
		}
		if = {
			limit = {
				is_gestalt = no
				NOT = {
					has_same_ethic_as_ideology = event_target:ideology_target
				}
			}
			event_target:ideology_target = {
				switch = {
					trigger = has_ethic
					ethic_militarist = { root = { shift_ethic_from_adopt_ideology = { ETHIC = militarist OPPOSITE = pacifist } } }
					ethic_pacifist = { root = { shift_ethic_from_adopt_ideology = { ETHIC = pacifist OPPOSITE = militarist } } }
					ethic_xenophobe = { root = { shift_ethic_from_adopt_ideology = { ETHIC = xenophobe OPPOSITE = xenophile } } }
					ethic_xenophile = { root = { shift_ethic_from_adopt_ideology = { ETHIC = xenophile OPPOSITE = xenophobe } } }
					ethic_materialist = { root = { shift_ethic_from_adopt_ideology = { ETHIC = materialist OPPOSITE = spiritualist } } }
					ethic_spiritualist = { root = { shift_ethic_from_adopt_ideology = { ETHIC = spiritualist OPPOSITE = materialist } } }
					ethic_authoritarian = { root = { shift_ethic_from_adopt_ideology = { ETHIC = authoritarian OPPOSITE = egalitarian } } }
					ethic_egalitarian = { root = { shift_ethic_from_adopt_ideology = { ETHIC = egalitarian OPPOSITE = authoritarian } } }
				}
			}
		}
		if = {
			limit = {
				has_modifier = unified_ideology_not_accomplished
				has_overlord_ideology = yes
			}
			remove_modifier = unified_ideology_not_accomplished
		}
		if = {
			limit = {
				has_modifier = imposed_ideology_not_accomplished
				has_country_flag = imposed_ideology_@event_target:ideology_target
			}
			remove_modifier = imposed_ideology_not_accomplished
			every_ideology = {
				prev = { remove_country_flag = imposed_ideology_@prev }
			}
		}
		remove_country_flag = adopt_ideology_needs_adoption
		remove_country_flag = adopt_ideology_needs_ethic_shift
		remove_country_flag = adopt_ideology_yourself
	}

	# Delayed because the aborted project is still listed while this runs
	on_cancel = {
		country_event = { id = forceofwill.1050 days = 1 }
	}

	on_fail = {
		remove_modifier = imposed_ideology_not_accomplished
		remove_country_flag = imposed_ideology_@event_target:ideology_target
		remove_country_flag = adopt_ideology_yourself
	}

	abort_trigger = {
		OR = {
			NOT = { exists = event_target:ideology_target }
			AND = {
				ideology? = { is_same_value = event_target:ideology_target }
				has_same_ethic_as_ideology = event_target:ideology_target
			}
		}
	}

	fail_trigger = {
		OR = {
			has_country_flag = under_ideological_ban_law
			AND = {
				has_country_flag = under_unified_ideology_law
				NOT = {
					overlord.ideology? = {
						is_same_value = event_target:ideology_target
					}
				}
			}
		}
	}
}
