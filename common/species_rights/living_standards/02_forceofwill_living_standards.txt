living_standard_native_supremacy = {
	potential = {
		has_force_of_will_dlc = yes
		from? = {
			has_ethic = ethic_absolute_xenophobe
		}
	}

	allow = {
		hidden_trigger = { exists = from }
		custom_tooltip = {
			fail_text = too_alien
			is_same_species = from.species
		}
		custom_tooltip = {
			fail_text = SPECIES_MUST_NOT_BE_ASSIMILATING
			NOT = {
				has_citizenship_type = {
					type = citizenship_assimilation
					country = from
				}
			}
		}
	}

	triggered_pop_group_modifier = {
		potential = {
			NAND = {
				exists = owner
				owner = { has_deficit = consumer_goods }
			}
		}
		pop_cat_ruler_happiness = @living_standard_happiness_native_supremacy
		pop_cat_specialist_happiness = @living_standard_happiness_native_supremacy
		pop_cat_worker_happiness = @living_standard_happiness_native_supremacy
		pop_cat_civilian_happiness = @living_standard_happiness_native_supremacy
		pop_happiness = 0.1
	}

	pop_group_modifier = {
		pop_category_rulers_consumer_goods_upkeep_add = 1.5
		pop_category_specialists_consumer_goods_upkeep_add = 0.75
		pop_category_workers_consumer_goods_upkeep_add = 0.5
		pop_category_civilians_consumer_goods_upkeep_add = 0.5

		pop_cat_ruler_political_power = 10
		pop_cat_specialist_political_power = 5
		pop_cat_worker_political_power = 3
		pop_cat_civilian_political_power = 3

		pop_bonus_workforce_mult = 0.1
		bonus_pop_growth = 0.1
	}

	ai_will_do = {
		factor = 10
	}
}
