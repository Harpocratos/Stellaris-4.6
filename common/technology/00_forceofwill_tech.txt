tech_galactic_halls = {
	cost = @tier2cost1
	area = society
	tier = 2
	category = { statecraft }
	weight = @tier2weight1

	potential = {
		has_force_of_will_dlc = yes
		OR = {
			exists = ideology
			is_country_type = rebel # since when copying techs, rebels don't have an ideology yet
		}
	}

	prereqfor_desc = {
		custom = {
			title = tech_galactic_halls_unlock_tt
			desc = tech_galactic_halls_unlock_details
		}
	}

	weight_modifier = {
		modifier = {
			exists = ideology
			is_fanatic = yes
			mult = 2
		}
    }

	ai_weight = {
		weight = 1
		modifier = {
			add = 10
			any_owned_colony = {
				is_affected_by_bad_ideology = { COUNTRY = root }
			}
		}
		modifier = {
			add = 1
			any_owned_planet = { is_planet_revolting = yes }
		}
		modifier = {
			add = 5
			ideology? = { is_ideology_custodian = prev }
		}
		modifier = {
			add = 1
			ideological_momentum < @ideology_momentum_low
		}
		modifier = {
			add = 5
			ideological_momentum < @ideology_momentum_moderately_low
		}
		modifier = {
			add = 5
			ideological_momentum < @ideology_momentum_dangerously_low
		}
	}
}
