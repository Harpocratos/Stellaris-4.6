# Have Defensive Pacts or federation - from on_yearly_pulse_country
country_event = {
	id = ambition.150
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_ascension_perk = ap_defender_of_the_galaxy_nomads
		OR = {
			has_federation = yes
			any_playable_country = {
				OR = {
					has_defensive_pact = root
					has_association_status = root
				}
			}
		}
	}

	immediate = {
		#check for federation
		if = {
			limit = {
				has_federation = yes
			}
			complete_crisis_objective = crisobj_federated
		}
		#check for defensive pacts
		every_playable_country = {
			limit = {
				OR = {
					has_defensive_pact = root
					has_association_status = root
				}
			}
			root = {
				complete_crisis_objective = crisobj_defensive_pacts
			}
		}
	}
}

# Destroy Crisis Ships
# This = owner of ship 1 (combatant)
# From = owner of ship 2 (destroyed)
# FromFrom = ship 1
# FromFromFrom = ship 2
country_event = {
	id = ambition.180
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_ascension_perk = ap_defender_of_the_galaxy_nomads

		from = {
			OR = {
				is_crisis_faction = yes
				has_country_flag = declared_crisis
			}
		}
		fromfrom = {
			OR = {
				NOT = { is_ship_class = shipclass_starbase }
				fleet = { is_mobile = yes } #juggernauts
			}
		}
	}

	immediate = {
		owner = {
			complete_crisis_objective = crisobj_destroy_crisis_ships
		}
	}
}
