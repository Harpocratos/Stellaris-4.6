# Spawn Zroni Storm (when a Storm Caster starbase building is complete)
system_event = {
	id = ancrel.9900
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		NOT = { has_modifier = zroni_storm }
		any_starbase_in_system = {
			has_zroni_storm_caster = yes
		}
	}

	immediate = {
		random_starbase_in_system = {
			limit = {
				has_zroni_storm_caster = yes
			}
			save_event_target_as = storm_caster_starbase
			owner? = {
				save_event_target_as = storm_caster_empire
			}
		}
		if = {
			limit = {
				event_target:storm_caster_empire = {
					NOT = { has_deficit = sr_zro }
				}
			}
			add_modifier = {
				modifier = zroni_storm
				days = -1
			}
			star = {
				create_ambient_object = {
					type = "space_storm_1"
					location = this
				}
			}
			last_created_ambient_object = {
				set_ambient_object_flag = zroni_storm
				set_location = {
					target = prev
					distance = 0
					angle = random
				}
			}
			if = {
				limit = {
					event_target:storm_caster_empire = {
						has_ascension_perk = ap_archaeoengineers
					}
				}
				event_target:storm_caster_starbase = {
					add_starbase_component = {
						component = "STARBASE_AURA_ZRONI_STORM"
					}
				}
			}
		}
	}
}

# Remove Zroni Storm (when a Storm Caster starbase building is dismantled)
system_event = {
	id = ancrel.9905
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		has_modifier = zroni_storm
	}

	immediate = {
		remove_modifier = zroni_storm
		every_ambient_object = {
			limit = { has_ambient_object_flag = zroni_storm }
			destroy_ambient_object = this
		}
		if = {
			limit = {
				any_ship_in_system = {
					is_ship_class = shipclass_starbase
					has_component = "STARBASE_AURA_ZRONI_STORM"
				}
			}
			starbase = {
				remove_starbase_component = {
					component = "STARBASE_AURA_ZRONI_STORM"
				}
			}
		}
	}
}

# Fake Building Faculty of Archaeo studies (when research is completed)
country_event = {
	id = ancrel.9915
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		is_ai = yes
		last_increased_tech = tech_archaeostudies
	}

	immediate = {
		random_list = {
			75 = {}
			25 = {
				random_owned_colony = {
					limit = {
						has_building = building_research_lab_1
					}
					set_timed_carrier_flag = {
						flag = ignore_ai_building_limitations
						days = 1
					}
					remove_building = building_research_lab_1
					add_building = building_archaeostudies_faculty
				}
				modifier = {
					factor = 2
					has_ethic = ethic_materialist
				}
				modifier = {
					factor = 3
					has_ethic = ethic_fanatic_materialist
				}
				modifier = {
					factor = 4
					has_ethic = ethic_absolute_materialist
				}
				modifier = {
					factor = 5
					count_archaeological_site = {
						count > 3
						limit = {
							is_owned_by = root
							is_site_completed = yes
						}
					}
				}
				modifier = {
					factor = 2
					resource_stockpile_compare = {
						resource = minor_artifacts
						value > 300
					}
				}
				modifier = {
					factor = 2
					is_xenophile = yes
				}
				modifier = {
					factor = 0
					is_xenophobe = yes
				}
				modifier = {
					factor = 0.25
					is_spiritualist = yes
				}
			}
		}
	}
}
