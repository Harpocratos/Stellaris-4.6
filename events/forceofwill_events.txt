# Ideology: update Galactic Halls emitter status
# this = new owner
# from = megastructure
# fromfrom = old owner (if existing)
country_event = {
	id = forceofwill.15
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		from = { is_megastructure_galactic_halls = yes }
	}

	immediate = {
		set_reach_update_batch = begin
		if = {
			limit = {
				count_owned_megastructure = {
					limit = { is_megastructure_galactic_halls = yes }
					count > 1
				}
			}
			from = { on_galactic_hall_deleted = yes }
			delete_megastructure = from
		}
		else_if = {
			limit = { exists = ideology }
			from = {
				if = {
					limit = {
						NOT = { galactic_halls_have_same_ethic_as_ideology = { IDEOLOGY = root.ideology } }
					}
					respawn_galactic_halls = {
						IDEOLOGY = root.ideology
						OWNER = root
						PLANET = planet
					}
				}
				else = { # respawn_galactic_halls already adds as emitter hence the else
					delete_ideology_emitter = yes # to remove it from previous owner's ideology
					add_emitter_to_ideology = {
						target = root.ideology
						owner = root
					}
				}
			}
		}
		else = {
			from = { delete_ideology_emitter = yes }
		}
		set_reach_update_batch = end
	}
}

# This = Country
# Immigration bureau logic
country_event = {
	id = forceofwill.100
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_open_stars_sovereignty_empire = yes
		any_owned_starbase = { has_starbase_building = immigration_bureau }
	}

	immediate = {
		# For each Immigration bureau
		every_owned_starbase = {
			limit = { has_starbase_building = immigration_bureau }

			# Try to select a species to generate
			if = {
				limit = { is_immigration_bureau_valid = yes }

				solar_system = {
					# For each Adjacent empire
					every_country_neighbor_to_system = {
						limit = {
							is_immigration_bureau_neighbor_valid = {
								OWNER = root
							}
							any_owned_pop_group = {
								can_pop_group_generate = yes
							}
						}

						# Select a random species
						random_owned_pop_group = {
							limit = { can_pop_group_generate = yes }
							species = {
								save_event_target_as = migrating_species
							}
						}

						# Save the current wanted pops amount in the system
						prev = {
							if = {
								limit = { is_variable_set = bureau_num_pops@prev }
								change_variable = {
									which = bureau_num_pops@prev
									value = value:calc_immigration_bureau_pops_per_neighbor|OWNER|root|NEIGHBOR|prev|
								}
							}
							else = {
								set_variable = {
									which = bureau_num_pops@prev
									value = value:calc_immigration_bureau_pops_per_neighbor|OWNER|root|NEIGHBOR|prev|
								}
							}
						}

						if = {
							limit = {
								# If we can add a pop
								prev = {
									check_variable = {
										which = bureau_num_pops@prev
										value >= 1
									}
								}
							}

							# Round wanted pops
							set_variable = {
								which = local_new_pops
								value = prev.bureau_num_pops@prev
							}
							floor_variable = local_new_pops

							# Populate one of our planets with the found species
							root = {
								ordered_owned_planet = {
									position = 0
									# Limits inside priority to avoid gaining no pop
									order_by = value:immigration_bureau_planet_priority
									random_owned_pop_group = {
										limit = { is_same_species = event_target:migrating_species }
										add_pop_amount = local_new_pops
										root = { set_country_flag = immigration_bureau_pop_group_found }
									}
									if = {
										limit = { NOT = { root = { has_country_flag = immigration_bureau_pop_group_found } } }
										create_pop_group = {
											size = local_new_pops
											species = event_target:migrating_species
										}
									}
								}
								remove_country_flag = immigration_bureau_pop_group_found
							}

							# Remove added pops from wanted pops
							prev = {
								subtract_variable = {
									which = bureau_num_pops@prev
									value = local_new_pops
								}
							}
							clear_variable = local_new_pops
						}
					}
				}
			}

			# Fail safe if no species found
			if = {
				limit = {
					NOT = { exists = event_target:migrating_species }
				}
				if = {
					limit = {
						NOT = { has_starbase_flag = immigration_bureau_invalid }
					}
					set_starbase_flag = immigration_bureau_invalid
					create_message = {
						type = MESSAGE_TYPE_IMMIGRATION_BUREAU_SHUTDOWN
						localization = MESSAGE_TYPE_IMMIGRATION_BUREAU_SHUTDOWN_DESC
						days = 30
						target = this
						variable = {
							type = name
							localization = SYSTEM
							scope = solar_system
						}
					}
				}
			}
			else = {
				if = {
					limit = { has_starbase_flag = immigration_bureau_invalid }
					remove_starbase_flag = immigration_bureau_invalid
				}
			}
		}
	}
}

# Fired on war start and after every fleet battle.
# Re-evaluates the Wartime Modifier from scratch and refreshes its lifespan.
country_event = {
	id = forceofwill.300
	hide_window = yes
	is_triggered_only = yes

	# The modifier checks catch an empire that dropped the civic while a modifier was still running
	trigger = {
		OR = {
			is_warmongers_empire = yes
			has_modifier = warmongers_at_war_1
			has_modifier = warmongers_at_war_2
			has_modifier = warmongers_at_war_3
		}
	}

	immediate = {
		remove_modifier = warmongers_at_war_1
		remove_modifier = warmongers_at_war_2
		remove_modifier = warmongers_at_war_3

		if = {
			limit = {
				is_warmongers_empire = yes
				is_at_war = yes
			}
			set_variable = {
				which = local_modifier_type
				value = 0
			}

			# Find the strongest opponent and assign the matching modifier tier
			every_relation = {
				limit = { is_at_war_with = root }

				# Overwhelming adversary
				if = {
					limit = {
						relative_power = {
							who = root
							category = fleet
							value = overwhelming
						}
					}
					root = {
						set_variable = {
							which = local_modifier_type
							value = 3
						}
					}
				}

				# Superior adversary
				if = {
					limit = {
						root = {
							check_variable = { which = local_modifier_type value < 2 }
						}
						relative_power = {
							who = root
							category = fleet
							value = superior
						}
					}
					root = {
						set_variable = {
							which = local_modifier_type
							value = 2
						}
					}
				}
				# Equivalent adversary
				else_if = {
					limit = {
						root = {
							check_variable = { which = local_modifier_type value = 0 }
						}
						relative_power = {
							who = root
							category = fleet
							value = equivalent
						}
					}
					root = {
						set_variable = {
							which = local_modifier_type
							value = 1
						}
					}
				}
			}

			if = {
				limit = { check_variable = { which = local_modifier_type value = 3 } }
				add_modifier = { modifier = warmongers_at_war_3 days = @warmongers_at_war_modifier_duration }
			}
			else_if = {
				limit = { check_variable = { which = local_modifier_type value = 2 } }
				add_modifier = { modifier = warmongers_at_war_2 days = @warmongers_at_war_modifier_duration }
			}
			else_if = {
				limit = { check_variable = { which = local_modifier_type value = 1 } }
				add_modifier = { modifier = warmongers_at_war_1 days = @warmongers_at_war_modifier_duration }
			}

			clear_variable = local_modifier_type
		}
	}
}

country_event = {
	id = forceofwill.1095
	title = forceofwill.1095.name
	desc = forceofwill.1095.desc
	picture = {
		trigger = { from = { is_spiritualist = no } }
		picture = GFX_evt_raised_fist
	}
	picture = {
		trigger = { from = { is_spiritualist = yes } }
		picture = GFX_evt_spiritualist_ideology
	}
	show_sound = event_ideology_events_political_debate
	is_triggered_only = yes

	immediate = {
		from = {
			set_ideology_flag = ideology_clash_event_fired_@root
			save_event_target_as = ideology_clash_ideology
			ordered_system_within_emitter_reach = {
				limit = {
					is_within_ideology_reach = root.ideology
					any_ideology_emitter_country = {
						ideology? = { is_same_value = from }
						has_communications = root
					}
				}
				position = 0
				order_by = value:ideology_reach_left|IDEOLOGY|from|
				random_ideology_emitter_country = {
					limit = {
						ideology? = { is_same_value = from }
						has_communications = root
					}
					save_event_target_as = ideology_clash_emitter
				}
			}
		}
		set_timed_country_flag = {
			flag = ideology_clash_event_cooldown
			years = 20
		}
		set_variable = {
			which = ideology_clash_systems
			value = 0
		}
		set_variable = {
			which = ideology_clash_owned_systems
			value = 0
		}
		ideology = {
			every_system_within_emitter_reach = {
				limit = { is_within_ideology_reach = from }
				save_event_target_as = ideology_clash_system
				root = {
					change_variable = {
						which = ideology_clash_systems
						value = 1
					}
				}
				if = {
					limit = { is_owned_by = root }

					NOT = { exists = event_target:ideology_clash_emitter }
					can_control_access_for = event_target:ideology_clash_emitter
				}
			}
		}
		if = {
			limit = { exists = event_target:ideology_clash_emitter }
			set_closed_borders = {
				who = event_target:ideology_clash_emitter
				status = yes
			}
		}
		add_modifier = {
			modifier = foreign_influence_resistance
			years = 10
		}
	}

	option = {
		name = forceofwill.1095.b
		custom_tooltip = forceofwill.1095.b.tt
		allow = {
			custom_tooltip = {
				fail_text = this_must_be_ideology_custodian
				ideology? = { is_ideology_custodian = root }
			}
			custom_tooltip = {

			}

country_event = {
	id = forceofwill.1101
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		is_ai = yes
		years_passed >= 20
		has_force_of_will_dlc = yes
		is_gestalt = no
		can_initiate_ideology = yes
		can_found_ideology_from_ai_personality = yes
		NOT = {
			any_ideology = {
				root = { has_same_ethic_as_ideology = prev }
			}
		}
		# Either first time triggering this event or if a faction can be advocated
		OR = {
			NOT = { has_country_flag = ai_first_time_trying_ideology }
			AND = {
				OR = {
					is_subject = no
					NOR = {
						any_agreement = { agreement_preset = preset_dominion }
					}
				}
				fanatic_faction_exists = yes
			}
		}
	}

	immediate = {
		random = {
			chance = value:ai_roll_generate_ideology
			if = {
				limit = {
					has_country_flag = ai_first_time_trying_ideology
					can_found_ideology_from_any_faction = yes
					fanatic_faction_exists = yes
				}
				random_pop_faction = {
					limit = { is_faction_fanatic = yes }
					set_pop_faction_flag = ai_triggered_advocate
				}
			}
			else = {
				country_event = { id = forceofwill.1100 }
			}
		}
		set_country_flag = ai_first_time_trying_ideology
	}
}

	id = forceofwill.1112
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		is_ai = yes
		has_force_of_will_dlc = yes
		is_country_type = default
		NOT = { exists = ideology }
	}

	immediate = {
		every_revealed_ideology = {
			limit = {
				root = {
					foreign_ideology_affinity = {
						ideology = prev
						value > 5
					}
				}
			}
			save_event_target_as = tested_ideology
			root = {
				random = {
					chance = value:ai_roll_adopt_ideology|SCOPE|event_target:tested_ideology|
					set_country_flag = ai_should_adopt_ideology@event_target:tested_ideology
				}
				break = yes
			}
		}
	}
}

# Adjust Gestalt deviancy from foreign ideology and ideological strength multiplier
country_event = {
	id = forceofwill.1220
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		has_force_of_will_dlc = yes
		is_gestalt = yes
	}

	immediate = {
		set_variable = {
			which = ideological_strength_mult
			value = 0
		}
		every_owned_colony = {
			if = {
				# Foreign ideology on our colony OR cultural envoy converting
				limit = {
					OR = {
						ideology? = {
							OR = {
								NOT = { exists = root.ideology }
								NOT = { is_same_value = root.ideology }
							}
							any_adopting_country = {
								has_communications = root
							}
						}
						carrier = {
							any_fleet_in_orbit = {
								has_fleet_flag = cultural_envoy_fleet_converting
								controller = { NOT = { is_same_value = root } }
							}
						}
					}
				}
				carrier = {
					if = {
						limit = { has_modifier = foreign_ideology_deviancy_decreasing }
						remove_modifier = foreign_ideology_deviancy_decreasing
					}
					set_variable = {
						which = foreign_ideology_deviancy_value
						value = value:increased_foreign_ideology_deviancy
					}
					add_modifier = {
						modifier = foreign_ideology_deviancy
						mult = foreign_ideology_deviancy_value
					}
				}
			}
			else_if = {
				limit = {
					carrier = {
						OR = {
							has_modifier = foreign_ideology_deviancy
							has_modifier = foreign_ideology_deviancy_decreasing
						}
					}
				}
				carrier = {
					if = {
						limit = { has_modifier = foreign_ideology_deviancy }
						remove_modifier = foreign_ideology_deviancy
					}
					set_variable = {
						which = foreign_ideology_deviancy_value
						value = value:decreased_foreign_ideology_deviancy
					}
					if = {
						limit = {
							check_variable = {
								which = foreign_ideology_deviancy_value
								value <= 0
							}
						}
						remove_modifier = foreign_ideology_deviancy_decreasing
						clear_variable = foreign_ideology_deviancy_value
					}
					else = {
						add_modifier = {
							modifier = foreign_ideology_deviancy_decreasing
							mult = foreign_ideology_deviancy_value
						}
					}
				}
			}

			if = {
				# Update ideological strength mult
				limit = { exists = root.ideology }
				if = {
					limit = {
						planet_crime < @ideology_low_deviancy_threshold
					}
					root = {
						change_variable = {
							which = ideological_strength_mult
							value = prev.value:calc_planet_ideological_strength_mult_low_deviancy
						}
					}
				}
				else = {
					root = {
						change_variable = {
							which = ideological_strength_mult
							value = prev.value:calc_planet_ideological_strength_mult_high_deviancy
						}
					}
				}
			}
		}
	}
}

# Gestalt AIs start abandoning their Ideology after two consecutive months of falling Strength, or two of widespread Deviancy
# This = Country
country_event = {
	id = forceofwill.1235
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		has_force_of_will_dlc = yes
		is_ai = yes
		is_country_type = default
		is_gestalt = yes
		exists = ideology
		num_owned_colonies > 0
		NOR = {
			has_country_flag = abandoning_ideology
			has_country_flag = recently_abandon_ideology
			has_country_flag = ideology_cannot_abandon_imposed
			has_country_flag = cant_abandon_ideology_forceofwill
		}
	}

	immediate = {
		set_variable = {
			which = empire_deviancy
			value = 0
		}
		every_owned_colony = {
			root = {
				change_variable = {
					which = empire_deviancy
					value = prev.trigger:planet_crime
				}
			}
		}
		divide_variable = {
			which = empire_deviancy
			value = trigger:num_owned_colonies
		}

		if = {
			limit = { last_ideology_increase < @ai_abandon_ideology_min_strength_change }
			if = {
				limit = { has_country_flag = ideology_abandon_strength_met }
				set_country_flag = ai_auto_abandoning_ideology
			}
			else = {
				set_country_flag = ideology_abandon_strength_met
			}
		}
		else = {
			remove_country_flag = ideology_abandon_strength_met
		}

		if = {
			limit = {
				check_variable = {
					which = empire_deviancy
					value > @ai_abandon_ideology_deviancy
				}
			}
			if = {
				limit = { has_country_flag = ideology_abandon_deviancy_met }
				set_country_flag = ai_auto_abandoning_ideology
			}
			else = {
				set_country_flag = ideology_abandon_deviancy_met
			}
		}
		else = {
			remove_country_flag = ideology_abandon_deviancy_met
		}

		if = {
			limit = { has_country_flag = ai_auto_abandoning_ideology }
			start_abandon_ideology_process = yes
		}

		clear_variable = empire_deviancy
	}
}

		has_force_of_will_dlc = yes

		years_passed > 10
	}

	immediate = {
		# Faction approval window logic
		every_pop_faction = {
			if = {
				limit = { NOT = { is_variable_set = fanatic_approval_months } }
				set_variable = {
					which = fanatic_approval_months
					value = 0
				}
			}
			if = {
				limit = { is_eligible_fanatic = yes }
				change_variable = {
					which = fanatic_approval_months
					value = 1
				}
			}
			else = {
				set_variable = {
					which = fanatic_approval_months
					value = 0
				}
			}
		}

		ordered_pop_faction = {
			limit = {
				is_eligible_fanatic = yes
				check_variable = {
					which = fanatic_approval_months
					value >= @fanatic_faction_approval_window
				}
			}
			order_by = trigger:faction_approval
			position = 0
			prev = {
				random_list = {
					1 = { # 0.1%: Faction becomes fanatic
						modifier = { # 0.5% while the empire has no fanatic faction yet
							add = 4
							fanatic_faction_exists = no
						}
						country_event = {
							id = forceofwill.2105
							scopes = { from = prev }
						}
					}
					999 = { # Nothing happens
						modifier = {
							add = -4
							fanatic_faction_exists = no
						}
					}
				}
			}
		}
	}
}

# Monthly pulse for Cultural Envoys
country_event = {
	id = forceofwill.7000
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_force_of_will_dlc = yes
		has_technology = tech_cultural_envoys
	}

	immediate = {
		save_event_target_as = envoy_country

		every_owned_fleet = {
			limit = {
				has_fleet_flag = cultural_envoy_fleet_converting
			}
			save_event_target_as = envoy_fleet

			# Target empire loses opinion of the envoy empire each month.
			# cultural_envoy_target is the colony the envoy is converting: a planet OR an arkship (ship colony).
			cultural_envoy_target? = {
				if = {
					limit = {
						is_colony = yes
						owner? = {
							NOT = {
								is_same_value = event_target:envoy_country
							}
						}
					}
					owner = {
						add_opinion_modifier = {
							who = event_target:envoy_country
							modifier = opinion_cultural_envoy_converting
							mult = event_target:envoy_fleet.value:cultural_envoy_conversion_modifier_mult
						}
					}
					if = {
						limit = {
							owner = {
								is_gestalt = no # Gestalt pops can't be converted
							}
						}
						set_variable = {
							which = pop_convertible_group_count
							value = value:cultural_envoy_convertible_group_count
						}
						if = {
							limit = {
								check_variable = {
									which = pop_convertible_group_count
									value > 0
								}
							}
							# Spread the monthly conversion across every group that still has pops to convert,
							# so capacity is not wasted on a single nearly-converted group.
							set_variable = {
								which = pop_conversion_amount
								value = event_target:envoy_fleet.value:cultural_envoy_conversion_amount
							}
							divide_variable = {
								which = pop_conversion_amount
								value = pop_convertible_group_count
							}
							ceiling_variable = pop_conversion_amount
							if = {
								limit = {
									event_target:envoy_country.ideology = {
										is_authoritarian = yes
									}
								}
								cultural_envoy_convert_pop_groups = {
									ETHIC = ethic_authoritarian
								}
							}
							else_if = {
								limit = {
									event_target:envoy_country.ideology = {
										is_egalitarian = yes
									}
								}
								cultural_envoy_convert_pop_groups = {
									ETHIC = ethic_egalitarian
								}
							}
							else_if = {
								limit = {
									event_target:envoy_country.ideology = {
										is_militarist = yes
									}
								}
								cultural_envoy_convert_pop_groups = {
									ETHIC = ethic_militarist
								}
							}
							else_if = {
								limit = {
									event_target:envoy_country.ideology = {
										is_pacifist = yes
									}
								}
								cultural_envoy_convert_pop_groups = {
									ETHIC = ethic_pacifist
								}
							}
							else_if = {
								limit = {
									event_target:envoy_country.ideology = {
										is_materialist = yes
									}
								}
								cultural_envoy_convert_pop_groups = {
									ETHIC = ethic_materialist
								}
							}
							else_if = {
								limit = {
									event_target:envoy_country.ideology = {
										is_spiritualist = yes
									}
								}
								cultural_envoy_convert_pop_groups = {
									ETHIC = ethic_spiritualist
								}
							}
							else_if = {
								limit = {
									event_target:envoy_country.ideology = {
										is_xenophobe = yes
									}
								}
								cultural_envoy_convert_pop_groups = {
									ETHIC = ethic_xenophobe
								}
							}
							else_if = {
								limit = {
									event_target:envoy_country.ideology = {
										is_xenophile = yes
									}
								}
								cultural_envoy_convert_pop_groups = {
									ETHIC = ethic_xenophile
								}
							}
							clear_variable = pop_conversion_amount

							# Holds converted pops in the envoy's faction until this colony goes long enough without a conversion
							set_timed_planet_flag = {
								flag = recently_converted_by_ideology@event_target:envoy_country.ideology
								days = @cultural_envoy_conversion_attraction_days
							}
						}
						clear_variable = pop_convertible_group_count
					}
				}
			}

			# Leader XP grant on the envoy fleet's leader.
			leader? = {
				add_experience = value:cultural_envoy_leader_experience
			}
		}
	}
}

# A Cultural Envoy fleet was destroyed by another empire while actively converting a colony.
# Triggered from on_fleet_destroyed_victim.
# This = envoy country (owner of destroyed fleet)
# From = offender country (owner of the killing fleet)
# FromFrom = destroyed envoy fleet
# FromFromFrom = offender fleet
country_event = {
	id = forceofwill.7025
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_force_of_will_dlc = yes
		exists = ideology
		fromfrom = {
			has_fleet_flag = cultural_envoy_fleet_converting
			exists = orbit
			orbit = {
				is_colony = yes
				exists = owner
				owner = {
					NOT = { is_same_value = root }
				}
			}
		}
	}

	immediate = {
		save_event_target_as = envoy_country
		from = { save_event_target_as = offender_country }
		fromfrom.orbit = {
			save_event_target_as = converting_colony
		}

		# Each destroyed envoy adds a stack to cultural_envoys_martyred.
		change_variable = { which = cultural_envoys_martyred value = 1 }
		remove_modifier = cultural_envoy_martyr
		add_modifier = {
			modifier = cultural_envoy_martyr
			mult = cultural_envoys_martyred
			days = 3600
		}
		# Store modifier expiration date (required for modifier stacking)
		set_variable = {
			which = cultural_envoy_martyr_expiry
			value = trigger:days_passed
		}
		change_variable = {
			which = cultural_envoy_martyr_expiry
			value = 3600
		}
		country_event = { id = forceofwill.7030 days = 3600 }

		# Every empire sharing the envoy's ideology loses opinion of the offender.
		ideology = {
			every_adopting_country = {
				limit = {
					NOT = { is_same_value = event_target:offender_country }
				}
				add_opinion_modifier = {
					who = event_target:offender_country
					modifier = opinion_cultural_envoy_destroyed
				}
			}
		}

		# Reduce planet stability when killing a converting envoy
		if = {
			limit = {
				event_target:converting_colony = {
					owner = {
						OR = {
							NOT = { exists = ideology }
							ideology = {
								NOT = { is_same_value = event_target:envoy_country.ideology }
							}
						}
					}
					colony? = {
						pop_ideology_percentage = {
							percentage > 0.3
							ideology = event_target:envoy_country.ideology
						}
					}
				}
			}
			event_target:converting_colony = {
				add_modifier = {
					modifier = envoy_martyr_modifier
					years = 25
				}
			}
			event_target:envoy_country.ideology = {
				save_global_event_target_as = envoy_martyr_ideology@event_target:converting_colony
			}
		}

		# Cultural envoy destroyed event
		if = {
			limit = {
				NOT = { has_country_flag = first_cultural_envoy_destroyed_victim }
			}
			set_country_flag = first_cultural_envoy_destroyed_victim
			country_event = {
				id = forceofwill.7026
				scopes = {
					from = from
					fromfrom = fromfrom
					fromfromfrom = event_target:converting_colony
				}
			}
		}
		else = {
			create_message = {
				type = MESSAGE_CULTURAL_ENVOY_DESTROYED
				localization = MESSAGE_CULTURAL_ENVOY_DESTROYED_DESC
				days = 30
				target = event_target:converting_colony
				variable = {
					type = name
					localization = CULTURAL_ENVOY_COUNTRY
					scope = root
				}
				variable = {
					type = name
					localization = SYSTEM_OF_CONVERTING_COLONY
					scope = event_target:converting_colony.solar_system
				}
			}
		}

		from = {
			if = {
				limit = {
					NOT = { has_country_flag = first_cultural_envoy_destroyed_perp }
				}
				set_country_flag = first_cultural_envoy_destroyed_perp
				country_event = {
					id = forceofwill.7027
					scopes = {
						from = root
						fromfrom = event_target:converting_colony
					}
				}
			}
			else = {
				create_message = {
					type = MESSAGE_FOREIGN_CULTURAL_ENVOY_DESTROYED
					localization = MESSAGE_FOREIGN_CULTURAL_ENVOY_DESTROYED_DESC
					days = 30
					target = event_target:converting_colony
					variable = {
						type = name
						localization = SYSTEM_OF_CONVERTING_COLONY
						scope = event_target:converting_colony.solar_system
					}
				}
			}
		}
	}
}

# Decays one stack of the cultural_envoy_martyr buff
country_event = {
	id = forceofwill.7030
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		set_variable = {
			which = cultural_envoy_martyr_days_left
			value = cultural_envoy_martyr_expiry
		}
		subtract_variable = {
			which = cultural_envoy_martyr_days_left
			value = trigger:days_passed
		}

		subtract_variable = { which = cultural_envoys_martyred value = 1 }
		remove_modifier = cultural_envoy_martyr
		if = {
			limit = {
				check_variable = { which = cultural_envoys_martyred value > 0 }
				check_variable = { which = cultural_envoy_martyr_days_left value > 0 }
			}
			add_modifier = {
				modifier = cultural_envoy_martyr
				days = 1
				time_multiplier = cultural_envoy_martyr_days_left
				mult = cultural_envoys_martyred
			}
		}
		else = {
			clear_variable = cultural_envoys_martyred
			clear_variable = cultural_envoy_martyr_expiry
		}
		clear_variable = cultural_envoy_martyr_days_left
	}
}

pop_faction_event = {
	id = forceofwill.7501
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_faction_fanatic = yes
		ideology? = {
			NOT = { is_ideology_custodian = prev.owner }
		}
		NOT = { has_pop_faction_flag = fanatic_faction_demand_federation_unlocked } # Last demand to unlock
		NOR = { # Every unlocked demand must be valid to unlock the next
			has_pop_faction_flag = fanatic_faction_demand_peace_cooldown
			has_pop_faction_flag = fanatic_faction_demand_borders_cooldown
			has_pop_faction_flag = fanatic_faction_demand_migration_cooldown
			has_pop_faction_flag = fanatic_faction_demand_adopt_cooldown
		}
	}

	immediate = {
		switch = {
			trigger = has_pop_faction_flag
			# Check to unlock federation demand
			fanatic_faction_demand_adopt_unlocked = {
				if = {
					limit = { has_pop_faction_flag = fanatic_faction_demand_adopt_fulfilled }
					random = {
						chance = @unlock_fanatic_faction_demand_chance
						set_pop_faction_flag = fanatic_faction_demand_federation_unlocked
					}
				}
			}
			# Check to unlock adopt Ideology demand
			fanatic_faction_demand_migration_unlocked = {
				if = {
					limit = { has_pop_faction_flag = fanatic_faction_demand_migration_fulfilled }
					random = {
						chance = @unlock_fanatic_faction_demand_chance
						set_pop_faction_flag = fanatic_faction_demand_adopt_unlocked
						set_timed_pop_faction_flag = {
							flag = fanatic_faction_demand_adopt_cooldown
							years = 2
						}
					}
				}
			}
			# Check to unlock migration pact demand
			fanatic_faction_demand_borders_unlocked = {
				if = {
					limit = { has_pop_faction_flag = fanatic_faction_demand_borders_fulfilled }
					random = {
						chance = @unlock_fanatic_faction_demand_chance
						set_pop_faction_flag = fanatic_faction_demand_migration_unlocked
						set_timed_pop_faction_flag = {
							flag = fanatic_faction_demand_migration_cooldown
							years = 2
						}
					}
				}
			}
			# Check to unlock open borders demand
			fanatic_faction_demand_peace_fulfilled = {
				random = {
					chance = @unlock_fanatic_faction_demand_chance
					set_pop_faction_flag = fanatic_faction_demand_borders_unlocked
					set_timed_pop_faction_flag = {
						flag = fanatic_faction_demand_borders_cooldown
						years = 2
					}
				}
			}
		}
	}
}

pop_faction_event = {
	id = forceofwill.7506
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_faction_fanatic = yes
		exists = owner
		ideology? = {
			exists = owner
			NOT = { is_ideology_custodian = prev.owner }
		}
	}

	immediate = {
		# Check peace demand
		if = {
			limit = {
				owner = { is_at_war_with = prev.ideology.owner }
			}
			set_timed_pop_faction_flag = {
				flag = fanatic_faction_demand_peace_cooldown
				years = 2
			}
			remove_pop_faction_flag = fanatic_faction_demand_peace_fulfilled
		}
		else = {
			set_pop_faction_flag = fanatic_faction_demand_peace_fulfilled
		}
		# Check open borders demand
		if = {
			limit = { has_pop_faction_flag = fanatic_faction_demand_borders_unlocked }
			if = {
				limit = {
					owner = { has_closed_borders = prev.ideology.owner }
				}
				set_timed_pop_faction_flag = {
					flag = fanatic_faction_demand_borders_cooldown
					years = 2
				}
				remove_pop_faction_flag = fanatic_faction_demand_borders_fulfilled
			}
			else = {
				set_pop_faction_flag = fanatic_faction_demand_borders_fulfilled
			}
		}
		# Check migration treaty demand
		if = {
			limit = { has_pop_faction_flag = fanatic_faction_demand_migration_unlocked }
			if = {
				limit = {
					NOT = {
						OR = {
							owner = { has_diplo_migration_treaty = prev.ideology.owner }
							ideology.owner = { has_diplo_migration_treaty = prev.owner }
						}
					}
				}
				set_timed_pop_faction_flag = {
					flag = fanatic_faction_demand_migration_cooldown
					years = 2
				}
				remove_pop_faction_flag = fanatic_faction_demand_migration_fulfilled
			}
			else = {
				set_pop_faction_flag = fanatic_faction_demand_migration_fulfilled
			}
		}
		# Check adopt ideology demand
		if = {
			limit = { has_pop_faction_flag = fanatic_faction_demand_adopt_unlocked }
			if = {
				limit = {
					owner = {
						OR = {
							NOT = { exists = ideology }
							ideology = {
								NOT = { is_same_value = prevprev.ideology }
							}
						}
					}
				}
				set_timed_pop_faction_flag = {
					flag = fanatic_faction_demand_adopt_cooldown
					years = 2
				}
				remove_pop_faction_flag = fanatic_faction_demand_adopt_fulfilled
			}
			else = {
				set_pop_faction_flag = fanatic_faction_demand_adopt_fulfilled
			}
		}
	}
}

country_event = {
	id = forceofwill.7510
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		fanatic_faction_exists = yes
		ideologies_enabled = yes
		is_country_type = default
	}

	immediate = {
		every_pop_faction = {
			limit = {
				is_faction_fanatic = yes
				is_undecided_cyber_creed_faction = no
				NOT = { has_pop_faction_flag = ideology_adoption_ultimatum_triggered }
			}
			if = {
				limit = {
					support >= 0.9
					OR = {
						root = { NOT = { exists = ideology } }
						NOT = { has_same_ethic_as_ideology = root.ideology }
					}
				}
				if = {
					limit = {
						NOT = { is_variable_set = approval_months_before_ultimatum }
					}
					set_variable = {
						which = approval_months_before_ultimatum
						value = 0
					}
				}
				change_variable = {
					which = approval_months_before_ultimatum
					value = 1
				}
				if = {
					limit = {
						check_variable = {
							which = approval_months_before_ultimatum
							value > 60 # 5 years
						}
					}
					set_pop_faction_flag = ideology_adoption_ultimatum_triggered
					clear_variable = approval_months_before_ultimatum
					root = {
						country_event = {
							id = forceofwill.7515
							scopes = { from = prev }
						}
					}
				}
			}
			else_if = {
				limit = { is_variable_set = approval_months_before_ultimatum }
				clear_variable = approval_months_before_ultimatum
			}
		}
