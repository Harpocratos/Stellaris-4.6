country_event = {
	id = arkship.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_nomadic = yes
	}

	immediate = {
		random_list = {
			50 = {
				modifier = {	# For the first five years you get no events so you have some time to journey around
					factor = 0
					years_passed < 5
				}
				modifier = {	# We increase events to bring you into the mid-game
					factor = 1.5
					mid_game_years_passed < 1
					years_passed >= 5
				}
				modifier = {	# A little cooldown so you don't get an event right as you join a war
					factor = 0
					has_country_flag = nomad_recent_war
				}
				modifier = {	# Block the event pool if all Arkships are in combat.
					factor = 0
					NOT = {
						any_owned_ship = {
							is_arkship_ship = yes
							is_in_combat = no
						}
					}
				}
				generate_arkship_event = yes
			}
			100 = { # No Arkship Event today
				country_event = { id = arkship.1 days = @very_short_arkship_event_delay_days random = @very_short_arkship_event_delay_random } #Roughly 3-5 year cooldown to try and fire another event
			}
		}
		country_event = { id = nomads.7000 }
	}
}

				count > 1
				limit = {
					is_country_type = default

fleet_event = {
	id = arkship.860
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		owner? = {
			is_nomadic = yes
			NOT = {
				any_owned_contract = {
					issuer = {
						is_same_value = fromfrom.space_owner
					}
				}
			}
		}
		NOT = { has_fleet_flag = supress_nomad_border_notification }
		NOT = {
			from.space_owner? = { is_same_value = fromfrom.space_owner }
		}
		fromfrom.space_owner? = {
			NOT = { is_same_value = root.owner }
			OR = {
				# Either we've closed borders to them
				has_closed_borders = root.owner
				# Or recently expelled them
				root.owner = {
					has_truce = prev
					has_country_flag = expelled_from_@prev
				}
			}
			NOR = {
				is_same_value = root.owner
				AND = {
					is_fallen_empire_spiritualist = yes
					root.owner = { has_relic = r_severed_head }
				}
			}
		}
		OR = {
			is_cloaked = no
			AND = {
				is_cloaked = yes
				from.starbase? = {
					has_cloaking_detection >= root.trigger:has_cloaking_strength
				}
			}
		}
	}

	immediate = {
		root.owner = { save_event_target_as = breaching_nomad }
		fromfrom.space_owner = {
			if = {
				limit = {
					NOT = { has_country_flag = arkship_breached_closed_borders_recently@event_target:breaching_nomad }
				}
				set_timed_country_flag = {
					flag = arkship_breached_closed_borders_recently@event_target:breaching_nomad
					years = 3
				}
				if = {
					limit = {
						NOT = { has_country_flag = arkship_crossing_closed_borders@event_target:breaching_nomad }
					}
					set_country_flag = arkship_crossing_closed_borders@event_target:breaching_nomad
					country_event = { id = arkship.865 }
				}
				else = {
					owner = {
						create_message = {
							type = MESSAGE_ARKSHIP_BREACHED_BORDERS
							localization = MESSAGE_ARKSHIP_BREACHED_BORDERS_DESC
							days = @toast_message_days
							target = from
						}
					}
				}
				add_opinion_modifier = {
					modifier = opinion_arkship_breaching_borders
					who = root.owner
				}
				# If you've recently been expelled, the opinion penalty is doubled.
				if = {
					limit = {
						root.owner = {
							has_truce = prev
							has_country_flag = expelled_from_@prev
						}
					}
					add_opinion_modifier = {
						modifier = opinion_arkship_breaching_borders
						who = root.owner
					}
				}
			}
		}
	}
}

					limit = { is_arkship_tier_1 = yes }
					last_created_ambient_object = {

					last_created_ambient_object = {
						set_variable = { which = arkship_debris_tier value = 2 }
					}

# Recover Disabled Nomad Vessel - special project success handler
# ROOT = rescuing empire (project owner); FROM = the disabled fleet
country_event = {
	id = arkship.1026
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { exists = from }
			# Arkship: regain control (reverses a seize; ownership is fleet-level), then re-enable.
			if = {
				limit = {
					from = {
						is_arkship_fleet = yes
						any_owned_ship = { is_arkship_ship = yes is_disabled = yes }
					}
				}
				from = {
					set_owner = root
					every_owned_ship = {
						limit = { is_arkship_ship = yes is_disabled = yes }
						repair_amount = 1000
						set_disabled = no
						remove_ship_flag = disabled_by_default_empire_event
					}
					set_controller = root	# end enemy occupation; set_owner alone leaves control with the occupier
				}
			}
			# Waystation: re-enable only (no ownership restore, per design).
			else_if = {
				limit = {
					from = { any_owned_ship = { is_waystation_ship = yes is_disabled = yes } }
				}
				from = {
					every_owned_ship = {
						limit = { is_waystation_ship = yes is_disabled = yes }
						repair_amount = 1000
						set_disabled = no
					}
				}
			}
			from = { remove_fleet_flag = nomad_recovery_project_active }
		}
	}
}

# Offer the recovery special project when a nomad arkship or waystation is disabled,
# if the owner is nomadic and currently at war.
# ROOT = disabled starbase, FROM = fleet that disabled it
starbase_event = {
	id = arkship.1027
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		OR = {
			is_arkship_starbase = yes
			is_waystation_starbase = yes
		}
		owner? = {
			is_nomadic = yes
			is_at_war = yes
		}
	}

	immediate = {
		owner = { save_event_target_as = nomad_recovery_owner }
		solar_system = {
			# The fleet flag guard ensures that if several vessels are disabled in the
			# same system, each on_starbase_disabled firing picks a different one (and
			# prevents a second project on a fleet that already has one).
			random_fleet_in_system = {
				limit = {
					owner = { is_same_value = event_target:nomad_recovery_owner }
					NOT = { has_fleet_flag = nomad_recovery_project_active }
					OR = {
						AND = {
							is_arkship_fleet = yes
							any_owned_ship = { is_arkship_ship = yes is_disabled = yes }
						}
						any_owned_ship = { is_waystation_ship = yes is_disabled = yes }
					}
				}
				set_fleet_flag = nomad_recovery_project_active
				enable_special_project = {
					name = RECOVER_DISABLED_NOMAD_VESSEL
					location = this				# the disabled fleet
					owner = event_target:nomad_recovery_owner
				}
			}
		}
	}
}

	id = arkship.1440

	desc = "arkship.1440.desc"
	picture = GFX_evt_space_funeral
	show_sound = event_ship_thrusters
	event_chain = star_crossed_lovers_chain
	trackable = yes

	is_triggered_only = yes

	option = {
		name = arkship.1440.a
		if = {
			limit = { exists = event_target:star_crossed_empire }
			custom_tooltip = arkship.1440.a.tooltip
			add_intel = {
				who = event_target:star_crossed_empire
				amount = 20
			}
		}
	}

	after = {
		hidden_effect = {
			country_event = {
				id = arkship.1
				days = @medium_arkship_event_delay_days
				random = @very_short_arkship_event_delay_random
			}
			end_event_chain = star_crossed_lovers_chain
		}
		hidden_effect = {
			if = {
				limit = { exists = event_target:star_crossed_survivor }
				event_target:star_crossed_survivor = {
					kill_leader = { show_notification = no }
				}
				clear_global_event_target = star_crossed_survivor
			}
			if = {
				limit = { exists = event_target:star_crossed_empire }
				clear_global_event_target = star_crossed_empire
			}
		}
	}
}

# Re-sync the arkship Galactic Hall propagation flag when ideology changes
country_event = {
	id = arkship.1455
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = has_arkship_galactic_hall_module
	}

	immediate = {
		if = {
			limit = {
				ideology? = {
					is_xenophobe = yes
				}
			}
			set_country_flag = has_hall_module_propagation
		}
		else = {
			remove_country_flag = has_hall_module_propagation
		}
	}
}
