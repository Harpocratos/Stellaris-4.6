@unrest_event_cooldown = 360

carrier_event = {
	id = unrest.50
	is_triggered_only = yes
	hide_window = yes

	pre_triggers = {
		has_owner = yes
		has_ground_combat = no
	}

	trigger = {
		NOT = { is_planet_class = pc_cosmogenesis_world }
		planet_stability < @stabilitylevel2
		pop_amount >= 1000
		has_orbital_bombardment = no
		exists = owner
		owner = {
			is_country_type = default
			NOR = {
				any_situation = { is_situation_type = situation_planetary_revolt }
				any_situation = { is_situation_type = situation_ideological_revolt }
				has_country_flag = had_recent_successful_rebellion
			}
		}
		NOR = {
			has_carrier_flag = 3_year_owner_change_flag
			has_carrier_flag = recent_revolt_attempt
		}
		is_doomsday_planet = no
		exists = controller
		controller = { is_same_value = root.owner }
		OR = {
			owner = { is_gestalt = yes }
			any_owned_species = { species_has_happiness_with_owner = root.owner }
		}
		NOT = {
			owner = {
				has_active_event = { unrest.55 }
			}
		}
		check_variable_arithmetic = { #will the revolt actually tick up?
			which = value:potential_revolt_situation_monthly_progress_estimation
			value > 0
		}
	}

	immediate = {
		set_timed_carrier_flag = {
			flag = unrest_timer
			days = 360
		}
		if = { #Getting an impossible to prevent revolt due to culture shock is a bit harsh. (Planet can start with no jobs, for instance)
			limit = { has_culture_shock = yes }
			carrier_event = { id = unrest.54 days = 1880 random = 180 } #at least 8 years after owner change (including the 3 years via 3_year_owner_change_flag)
		}
		else = {
			carrier_event = { id = unrest.54 days = 300 random = 59 } #definitely within the next year
		}
	}
}

carrier_event = {
	id = unrest.54
	is_triggered_only = yes
	hide_window = yes

	pre_triggers = {
		has_owner = yes
		has_ground_combat = no
	}

	trigger = {
		NOT = { is_planet_class = pc_cosmogenesis_world }
		planet_stability < @stabilitylevel2
		pop_amount >= 1000
		has_orbital_bombardment = no
		exists = owner
		owner = {
			is_country_type = default
			NOR = {
				any_situation = { is_situation_type = situation_planetary_revolt }
				any_situation = { is_situation_type = situation_ideological_revolt }
				has_country_flag = had_recent_successful_rebellion
			}
		}
		NOR = {
			has_carrier_flag = 3_year_owner_change_flag
			has_carrier_flag = recent_revolt_attempt
		}
		exists = controller
		controller = { is_same_value = root.owner }
		OR = {
			owner = { is_gestalt = yes }
			any_owned_species = { species_has_happiness_with_owner = root.owner }
		}
		check_variable_arithmetic = { #will the revolt actually tick up?
			which = value:potential_revolt_situation_monthly_progress_estimation
			value > 0
		}
	}

	immediate = {
		if = {
			limit = { #Extra safety because it's possible for two planets to fire this delayed event on the same day and it will check the trigger before either of them can fire (as of March 2022)
				owner = {
					NOR = {
						any_situation = { is_situation_type = situation_planetary_revolt }
						any_situation = { is_situation_type = situation_ideological_revolt }
						has_country_flag = had_recent_successful_rebellion
					}
				}
			}
			owner = { #Generally, one would expect the strongest planet to be the center of the revolt
				#Use case here is if you conquer a whole bunch of enemy planets at once.
				ordered_owned_planet = {
					position = 0
					order_by = value:revolt_focal_planet_weight
					limit = {
						has_carrier_flag = unrest_timer
						planet_stability < @stabilitylevel2
						has_orbital_bombardment = no
						has_ground_combat = no
						NOR = {
							has_carrier_flag = 3_year_owner_change_flag
							has_carrier_flag = recent_revolt_attempt
						}
						exists = controller
						controller = { is_same_value = root.owner }
						OR = {
							owner = { is_gestalt = yes }
							any_owned_species = { species_has_happiness_with_owner = root.owner }
						}
						check_variable_arithmetic = { #will the revolt actually tick up?
							which = value:potential_revolt_situation_monthly_progress_estimation
							value > 0
						}
					}
					carrier_event = { id = unrest.55 }
				}
			}
		}
	}
}

carrier_event = {
	id = unrest.55
	title = unrest.55.name
	desc = {

				NOT = { is_planet_class = pc_ark }
			}
		}
		text = unrest.55.desc.martyr
	}
	desc = {
		trigger = {
			event_target:new_situation = { has_situation_flag = drift_revolt }
			root = {
				has_modifier = envoy_martyr_modifier
				is_planet_class = pc_ark
			}
		}
		text = unrest.55.desc.martyr_ark
	}
	desc = {
		trigger = {
			event_target:new_situation = { has_situation_flag = schism_revolt }
		}
		text = unrest.55.desc.schism
	}
	desc = {
		trigger = {
			owner = { is_gestalt = no }
			event_target:new_situation = {
				NOR = {
					has_situation_flag = slave_revolt
					has_situation_flag = drift_revolt
					has_situation_flag = schism_revolt
				}
			}
		}
		text = unrest.55.desc.standard
	}
	desc = {
		trigger = {
			owner = { is_gestalt = no }
			event_target:new_situation = { has_situation_flag = slave_revolt }
		}
		text = unrest.55.desc.slave
	}
	desc = {
		trigger = {
			owner = { is_gestalt = yes }
			any_owned_pop_group = {
				is_sapient = yes
				NOT = { has_ethic = ethic_gestalt_consciousness }
			}
		}
		text = unrest.55.desc.gestalt_standard
	}
	desc = {
		trigger = {
			owner = { is_gestalt = yes }
			count_owned_pop_amount = {
				limit = {
					is_sapient = yes
					NOT = { has_ethic = ethic_gestalt_consciousness }
				}
				count = 0
			}
		}
		text = unrest.55.desc.gestalt_breakaway_gestalt
	}
	picture = GFX_evt_night_raid
	show_sound = {
		sound = advisor_notification_planet_in_revolt
		sound_is_advisor = yes
	}
	location = root
	is_triggered_only = yes
	situation = event_target:new_situation

	immediate = {
		owner = {
			if = {
				limit = { root.colony = { planet_should_start_ideological_revolt = yes } }
				start_situation = {
					type = situation_ideological_revolt
					target = root
					effect = {
						save_event_target_as = new_situation
						set_situation_flag = drift_revolt
					}
				}
				if = {
					limit = { root = { has_modifier = envoy_martyr_modifier } }
					event_target:envoy_martyr_ideology@root = {
						save_global_event_target_as = ideological_revolt_target_ideology@event_target:new_situation
						save_event_target_as = ideological_revolt_target_ideology
					}
				}
				else = {
					ordered_revealed_ideology = {
						position = 0
						order_by = root.colony.value:pop_ideology_percentage_value|IDEOLOGY|prevprev|
						limit = {
							NOT = { is_same_value = root.owner.ideology }
						}
						save_global_event_target_as = ideological_revolt_target_ideology@event_target:new_situation
						save_event_target_as = ideological_revolt_target_ideology
					}
				}
			}
			else_if = {
				limit = { root.colony = { planet_should_start_ideological_schism = yes } }
				start_situation = {
					type = situation_ideological_revolt
					target = root
					effect = {
						save_event_target_as = new_situation
						set_situation_flag = schism_revolt
					}
				}
				ideology = {
					save_global_event_target_as = ideological_revolt_target_ideology@event_target:new_situation
					save_event_target_as = ideological_revolt_target_ideology
				}
			}
			else = {
				start_situation = {
					type = situation_planetary_revolt
					target = root
					effect = {
						save_event_target_as = new_situation
						if = {
							limit = {
								root = {
									count_owned_pop_amount = {
										count >= 500
										limit = {
											is_enslaved = yes
											is_sapient = yes
											pop_has_happiness = yes
											happiness < 0.4
										}
									}
								}
							}
							set_situation_flag = slave_revolt
						}
					}
				}
			}
		}
	}

	option = {
		name = unrest.55.A
		if = {
			limit = {
				event_target:new_situation = {
					OR = {
						has_situation_flag = drift_revolt
						has_situation_flag = schism_revolt
					}
				}
			}
			tooltip = {
				owner = {
					start_situation = {
						type = situation_ideological_revolt
						target = root
					}
				}
			}
		}
		else = {
			tooltip = {
				owner = {
					start_situation = {
						type = situation_planetary_revolt
						target = root
					}
				}
			}
		}
	}
}
